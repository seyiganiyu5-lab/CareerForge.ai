   488→function estimateYearsOfExperience(workExperience?: ResumeContext['workExperience']): string {
   489→  if (!workExperience || workExperience.length === 0) return 'several';
   490→  let totalYears = 0;
   491→  for (const exp of workExperience) {
   492→    if (exp.startDate) {
   493→      const start = new Date(exp.startDate);
   494→      const end = exp.current ? new Date() : (exp.endDate ? new Date(exp.endDate) : new Date());
   495→      const diffMs = end.getTime() - start.getTime();
   496→      totalYears += Math.max(0, Math.round(diffMs / (365.25 * 24 * 60 * 60 * 1000)));
   497→    }
   498→  }
   499→  return totalYears > 0 ? String(totalYears) : 'several';
   500→}

   529→
   530→export async function POST(
   531→  request: NextRequest,
   532→  { params }: { params: Promise<{ id: string }> }
   533→) {
   534→  try {
   535→    // 1. Authentication
   536→    const auth = await getAuthUser();
   537→    if (!auth) {
   538→      return NextResponse.json(
   539→        { success: false, error: 'Unauthorized' },
   540→        { status: 401 }
   541→      );
   542→    }
   543→
   544→    // 2. Parse & validate request body
   545→    const body: AiAssistBody = await request.json();
   546→    const { section, action, content, customPrompt, tone, resumeContext, itemId, field } = body;
   547→
   548→    if (!section || !action) {
   549→      return NextResponse.json(
   550→        { success: false, error: 'Missing required fields: section and action' },
   551→        { status: 400 }
   552→      );
   553→    }
   554→
   555→    if (!VALID_SECTIONS.includes(section as Section)) {
   556→      return NextResponse.json(
   557→        { success: false, error: `Invalid section. Must be one of: ${VALID_SECTIONS.join(', ')}` },
   558→        { status: 400 }
   559→      );
   560→    }
   561→
   562→    if (!VALID_ACTIONS.includes(action as Action)) {
   563→      return NextResponse.json(
   564→        { success: false, error: `Invalid action. Must be one of: ${VALID_ACTIONS.join(', ')}` },
   565→        { status: 400 }
   566→      );
   567→    }
   568→
   569→    if (action === 'custom' && !customPrompt) {
   570→      return NextResponse.json(
   571→        { success: false, error: 'customPrompt is required when action is "custom"' },
   572→        { status: 400 }
   573→      );
   574→    }
   575→
   576→    // 3. Resolve resume ID from route params
   577→    const { id } = await params;
   578→
   579→    // 4. Ownership check
   580→    const resume = await db.resume.findUnique({
   581→      where: { id },
   582→    });
   583→
   584→    if (!resume || resume.user_id !== auth.userId) {
   585→      return NextResponse.json(
   586→        { success: false, error: 'Resume not found' },
   587→        { status: 404 }
   588→      );
   589→    }
   590→
   591→    // 5. Feature gating
   592→    const usage = await checkAndIncrementUsage(auth.userId, FEATURE_KEYS.AI_RESUME_BUILDER);
   593→    if (!usage.allowed) {
   594→      return NextResponse.json(
   595→        {
   596→          success: false,
   597→          error: usage.reason || 'limit_reached',
   598→          limit: usage.limit,
   599→          used: usage.used,
   600→          planName: usage.planName,
   601→        },
   602→        { status: 403 }
   603→      );
   604→    }
   605→
   606→    // 6. Build the AI prompt
   607→    const { systemPrompt, userMessage } = buildPrompt(section, action, body);
   608→
   609→    // 7. Call the AI SDK
   610→    const zai = await ZAI.create();
   611→    const completion = await zai.chat.completions.create({
   612→      messages: [
   613→        { role: 'assistant', content: systemPrompt },
   614→        { role: 'user', content: userMessage },
   615→      ],
   616→      thinking: { type: 'disabled' },
   617→    });
   618→
   619→    const aiResponse = completion.choices[0]?.message?.content?.trim();
   620→
   621→    if (!aiResponse) {
   622→      return NextResponse.json(
   623→        { success: false, error: 'Failed to generate suggestion' },
   624→        { status: 500 }
   625→      );
   626→    }
   627→
   628→    // 8. Save to AiHistory
   629→    await db.aiHistory.create({
   630→      data: {
   631→        resume_id: id,
   632→        user_id: auth.userId,
   633→        section,
   634→        action,
   635→        prompt: systemPrompt + '\n\n' + userMessage,
   636→        result: aiResponse,
   637→        original_text: content || '',
   638→      },
   639→    });
   640→
   641→    // 9. Return success
   642→    return NextResponse.json({
   643→      success: true,
   644→      result: aiResponse,
   645→      action,
   646→      section,
   647→      usage: {
   648→        limit: usage.limit,
   649→        used: usage.used + 1, // reflect the just-incremented count
   650→      },
   651→    });
   652→  } catch (error) {
   653→    console.error('[AI Assist] Error:', error);
   654→
   655→    // Handle ZAI-specific errors gracefully
   656→    if (error instanceof Error) {
   657→      const message = error.message.toLowerCase();
   658→      if (
   659→        message.includes('rate limit') ||
   660→        message.includes('quota') ||
   661→        message.includes('too many requests')
   662→      ) {
   663→        return NextResponse.json(
   664→          { success: false, error: 'AI service is temporarily unavailable. Please try again in a moment.' },
   665→          { status: 503 }
   666→        );
   667→      }
   668→
   669→      if (
   670→        message.includes('api key') ||
   671→        message.includes('authentication') ||
   672→        message.includes('unauthorized')
   673→      ) {
   674→        return NextResponse.json(
   675→          { success: false, error: 'AI service configuration error. Please contact support.' },
   676→          { status: 500 }
   677→        );
   678→      }
   679→
   680→      if (
   681→        message.includes('timeout') ||
   682→        message.includes('network') ||
   683→        message.includes('fetch')
   684→      ) {
   685→        return NextResponse.json(
   686→          { success: false, error: 'AI service request timed out. Please try again.' },
   687→          { status: 504 }
   688→        );
   689→      }
   690→    }
   691→
   692→    return NextResponse.json(
   693→      { success: false, error: 'Failed to generate suggestion' },
   694→      { status: 500 }
   695→    );
   696→  }
   697→}
