   136→function getScoreColor(score: number) {
   137→  if (score >= 80) return { text: 'text-emerald-500', bg: 'bg-emerald-500/10', stroke: '#10b981', label: 'Excellent' };
   138→  if (score >= 60) return { text: 'text-amber-500', bg: 'bg-amber-500/10', stroke: '#f59e0b', label: 'Good' };
   139→  if (score >= 40) return { text: 'text-orange-500', bg: 'bg-orange-500/10', stroke: '#f97316', label: 'Fair' };
   140→  return { text: 'text-red-500', bg: 'bg-red-500/10', stroke: '#ef4444', label: 'Needs Work' };
   141→}
   142→
   143→function getBarColor(score: number) {
   144→  if (score >= 80) return 'bg-emerald-500';
   145→  if (score >= 60) return 'bg-amber-500';
   146→  if (score >= 40) return 'bg-orange-500';
   147→  return 'bg-red-500';
   148→}
   149→
   150→function getBarTrackColor(score: number) {
   151→  if (score >= 80) return 'bg-emerald-500/20';
   152→  if (score >= 60) return 'bg-amber-500/20';
   153→  if (score >= 40) return 'bg-orange-500/20';
   154→  return 'bg-red-500/20';
   155→}

   183→// ─── Score Ring Component ─────────────────────────────────────────

   185→function ScoreRing({ score, size = 160, showLabel = true }: { score: number; size?: number; showLabel?: boolean }) {
   186→  const radius = (size - 16) / 2;
   187→  const circumference = 2 * Math.PI * radius;
   188→  const offset = circumference - (score / 100) * circumference;
   189→  const { text, bg, stroke, label } = getScoreColor(score);
   190→
   191→  return (
   192→    <div className="relative flex items-center justify-center">
   193→      <svg width={size} height={size} className="-rotate-90" role="img" aria-label={`ATS score: ${score} out of 100`}>
   194→        <circle cx={size / 2} cy={size / 2} r={radius} fill="none" stroke="currentColor" strokeWidth="10" className="text-muted/20" />
   195→        <motion.circle
   196→          cx={size / 2} cy={size / 2} r={radius} fill="none" stroke={stroke} strokeWidth="10"
   197→          strokeLinecap="round" strokeDasharray={circumference} initial={{ strokeDashoffset: circumference }}
   198→          animate={{ strokeDashoffset: offset }} transition={{ duration: 1.2, ease: 'easeOut', delay: 0.3 }}
   199→        />
   200→      </svg>
   201→      <div className="absolute flex flex-col items-center">
   202→        <motion.span
   203→          className={cn('text-4xl font-bold font-[family-name:var(--font-poppins)]', text)}
   204→          initial={{ opacity: 0, scale: 0.5 }} animate={{ opacity: 1, scale: 1 }}
   205→          transition={{ duration: 0.5, delay: 0.6 }}
   206→        >
   207→          {score}
   208→        </motion.span>
   209→        <span className="text-xs text-muted-foreground font-[family-name:var(--font-inter)]">/ 100</span>
   210→        {showLabel && (
   211→          <Badge variant="secondary" className={cn('mt-1 text-[10px] font-medium border-0', bg, text)}>
   212→            {label}
   213→          </Badge>
   214→        )}
   215→      </div>
   216→    </div>
   217→  );
   218→}

   256→function CategoryCardIcon({ label, text, bg }: { label: string; text: string; bg: string }) {
   257→  const lower = label.toLowerCase();
   258→  if (lower.includes('format') || lower.includes('structure')) return <LayoutList className={cn('w-4 h-4', text)} />;
   259→  if (lower.includes('keyword') || lower.includes('skill')) return <Target className={cn('w-4 h-4', text)} />;
   260→  if (lower.includes('grammar') || lower.includes('spelling')) return <SpellCheck className={cn('w-4 h-4', text)} />;
   261→  if (lower.includes('content') || lower.includes('experience')) return <FileText className={cn('w-4 h-4', text)} />;
   262→  if (lower.includes('section') || lower.includes('order')) return <BookOpen className={cn('w-4 h-4', text)} />;
   263→  if (lower.includes('action') || lower.includes('verb')) return <Zap className={cn('w-4 h-4', text)} />;
   264→  return <BarChart3 className={cn('w-4 h-4', text)} />;
   265→}
