   463→   463→    }
   464→   464→  }
   465→   465→
   466→   466→  // Override for grammar action on any section that wasn't handled specifically
   467→   467→  if (action === 'grammar' && section !== 'professionalSummary' && section !== 'workExperience' && section !== 'education' && section !== 'skills' && section !== 'projects' && section !== 'certifications' && section !== 'languages') {
   468→   468→    userMessage = `Check and fix grammar, spelling, repetition, weak wording, and passive voice. Return only the corrected text.\n\n${content || '(empty)'}`;
   469→   469→  }
   470→   470→
   471→   471→  // Override for keywords action on any section that wasn't handled specifically
   472→   472→  if (action === 'keywords' && section !== 'professionalSummary' && section !== 'workExperience' && section !== 'skills') {
   473→   473→    userMessage = `Suggest ATS-friendly keywords commonly found in job descriptions for a ${jobTitle}. Explain why each keyword is valuable. Format as: keyword - reason.`;
   474→   474→  }
   475→   475→
   476→   476→  return { systemPrompt, userMessage };
   477→   477→}

   483→   483→function estimateYearsOfExperience(workExperience?: ResumeContext['workExperience']): string {
   484→   484→  if (!workExperience || workExperience.length === 0) return 'several';
   485→   485→  let totalYears = 0;
   486→   486→  for (const exp of workExperience) {
   487→   487→    if (exp.startDate) {
   488→   488→      const start = new Date(exp.startDate);
   489→   489→      const end = exp.current ? new Date() : (exp.endDate ? new Date(exp.endDate) : new Date());
   490→   490→      const diffMs = end.getTime() - start.getTime();
   491→   491→      totalYears += Math.max(0, Math.round(diffMs / (365.25 * 24 * 60 * 60 * 1000)));
   492→   492→    }
   493→   493→  }
   494→   494→  return totalYears > 0 ? String(totalYears) : 'several';
   495→   495→}

   524→   524→
   525→   525→export async function POST(
   526→   526→  request: NextRequest,
   527→   527→  { params }: { params: Promise<{ id: string }> }
   528→   528→) {
   529→   529→  try {
   530→   530→    // 1. Authentication
   531→   531→    const auth = await getAuthUser();
   532→   532→    if (!auth) {
   533→   533→      return NextResponse.json(
   534→   534→        { success: false, error: 'Unauthorized' },
   535→   535→        { status: 401 }
   536→   536→      );
   537→   537→    }
   538→   538→
   539→   539→    // 2. Parse & validate request body
   540→   540→    const body: AiAssistBody = await request.json();
   541→   541→    const { section, action, content, customPrompt, tone, resumeContext, itemId, field } = body;
   542→   542→
   543→   543→    if (!section || !action) {
   544→   544→      return NextResponse.json(
   545→   545→        { success: false, error: 'Missing required fields: section and action' },
   546→   546→        { status: 400 }
   547→   547→      );
   548→   548→    }
   549→   549→
   550→   550→    if (!VALID_SECTIONS.includes(section as Section)) {
   551→   551→      return NextResponse.json(
   552→   552→        { success: false, error: `Invalid section. Must be one of: ${VALID_SECTIONS.join(', ')}` },
   553→   553→        { status: 400 }
   554→   554→      );
   555→   555→    }
   556→   556→
   557→   557→    if (!VALID_ACTIONS.includes(action as Action)) {
   558→   558→      return NextResponse.json(
   559→   559→        { success: false, error: `Invalid action. Must be one of: ${VALID_ACTIONS.join(', ')}` },
   560→   560→        { status: 400 }
   561→   561→      );
   562→   562→    }
   563→   563→
   564→   564→    if (action === 'custom' && !customPrompt) {
   565→   565→      return NextResponse.json(
   566→   566→        { success: false, error: 'customPrompt is required when action is "custom"' },
   567→   567→        { status: 400 }
   568→   568→      );
   569→   569→    }
   570→   570→
   571→   571→    // 3. Resolve resume ID from route params
   572→   572→    const { id } = await params;
   573→   573→
   574→   574→    // 4. Ownership check
   575→   575→    const resume = await db.resume.findUnique({
   576→   576→      where: { id },
   577→   577→    });
   578→   578→
   579→   579→    if (!resume || resume.user_id !== auth.userId) {
   580→   580→      return NextResponse.json(
   581→   581→        { success: false, error: 'Resume not found' },
   582→   582→        { status: 404 }
   583→   583→      );
   584→   584→    }
   585→   585→
   586→   586→    // 5. Feature gating
   587→   587→    const usage = await checkAndIncrementUsage(auth.userId, FEATURE_KEYS.AI_RESUME_BUILDER);
   588→   588→    if (!usage.allowed) {
   589→   589→      return NextResponse.json(
   590→   590→        {
   591→   591→          success: false,
   592→   592→          error: usage.reason || 'limit_reached',
   593→   593→          limit: usage.limit,
   594→   594→          used: usage.used,
   595→   595→          planName: usage.planName,
   596→   596→        },
   597→   597→        { status: 403 }
   598→   598→      );
   599→   599→    }
   600→   600→
   601→   601→    // 6. Build the AI prompt
   602→   602→    const { systemPrompt, userMessage } = buildPrompt(section, action, body);
   603→   603→
   604→   604→    // 7. Call the AI SDK
   605→   605→    const zai = await ZAI.create();
   606→   606→    const completion = await zai.chat.completions.create({
   607→   607→      messages: [
   608→   608→        { role: 'assistant', content: systemPrompt },
   609→   609→        { role: 'user', content: userMessage },
   610→   610→      ],
   611→   611→      thinking: { type: 'disabled' },
   612→   612→    });
   613→   613→
   614→   614→    const aiResponse = completion.choices[0]?.message?.content?.trim();
   615→   615→
   616→   616→    if (!aiResponse) {
   617→   617→      return NextResponse.json(
   618→   618→        { success: false, error: 'Failed to generate suggestion' },
   619→   619→        { status: 500 }
   620→   620→      );
   621→   621→    }
   622→   622→
   623→   623→    // 8. Save to AiHistory
   624→   624→    await db.aiHistory.create({
   625→   625→      data: {
   626→   626→        resume_id: id,
   627→   627→        user_id: auth.userId,
   628→   628→        section,
   629→   629→        action,
   630→   630→        prompt: systemPrompt + '\n\n' + userMessage,
   631→   631→        result: aiResponse,
   632→   632→        original_text: content || '',
   633→   633→      },
   634→   634→    });
   635→   635→
   636→   636→    // 9. Return success
   637→   637→    return NextResponse.json({
   638→   638→      success: true,
   639→   639→      result: aiResponse,
   640→   640→      action,
   641→   641→      section,
   642→   642→      usage: {
   643→   643→        limit: usage.limit,
   644→   644→        used: usage.used + 1, // reflect the just-incremented count
   645→   645→      },
   646→   646→    });
   647→   647→  } catch (error) {
   648→   648→    console.error('[AI Assist] Error:', error);
   649→   649→
   650→   650→    // Handle ZAI-specific errors gracefully
   651→   651→    if (error instanceof Error) {
   652→   652→      const message = error.message.toLowerCase();
   653→   653→      if (
   654→   654→        message.includes('rate limit') ||
   655→   655→        message.includes('quota') ||
   656→   656→        message.includes('too many requests')
   657→   657→      ) {
   658→   658→        return NextResponse.json(
   659→   659→          { success: false, error: 'AI service is temporarily unavailable. Please try again in a moment.' },
   660→   660→          { status: 503 }
   661→   661→        );
   662→   662→      }
   663→   663→
   664→   664→      if (
   665→   665→        message.includes('api key') ||
   666→   666→        message.includes('authentication') ||
   667→   667→        message.includes('unauthorized')
   668→   668→      ) {
   669→   669→        return NextResponse.json(
   670→   670→          { success: false, error: 'AI service configuration error. Please contact support.' },
   671→   671→          { status: 500 }
   672→   672→        );
   673→   673→      }
   674→   674→
   675→   675→      if (
   676→   676→        message.includes('timeout') ||
   677→   677→        message.includes('network') ||
   678→   678→        message.includes('fetch')
   679→   679→      ) {
   680→   680→        return NextResponse.json(
   681→   681→          { success: false, error: 'AI service request timed out. Please try again.' },
   682→   682→          { status: 504 }
   683→   683→        );
   684→   684→      }
   685→   685→    }
   686→   686→
   687→   687→    return NextResponse.json(
   688→   688→      { success: false, error: 'Failed to generate suggestion' },
   689→   689→      { status: 500 }
   690→   690→    );
   691→   691→  }
