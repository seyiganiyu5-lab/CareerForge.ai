   108→function analyzeContactInfo(personalInfo: PersonalInfo): { score: number; suggestions: Suggestion[] } {

   110→  const suggestions: Suggestion[] = [];
   111→
   112→  const hasEmail = !!personalInfo.email && personalInfo.email.trim().length > 0;

   119→  const hasTitle = !!personalInfo.professionalTitle && personalInfo.professionalTitle.trim().length > 0;

   127→      priority: 'high',
   128→      section: 'personalInfo',

   132→    score -= 25;

   134→      category: 'contactInfo',
   135→      message: 'Add a phone number so recruiters can reach you directly',
   136→      priority: 'high',
   137→      section: 'personalInfo',
   138→    });
   139→  }
   140→  if (!hasLocation) {
   141→    score -= 20;
   142→    suggestions.push({
   143→      category: 'contactInfo',
   144→      message: 'Add your location (city, state) — many employers filter by location',
   145→      priority: 'high',
   146→      section: 'personalInfo',
   147→    });

   191→function analyzeFormatting(resume: ResumeData): { score: number; issues: FormattingIssue[] } {
   192→  let score = 100;
   193→  const issues: FormattingIssue[] = [];
   194→
   195→  // Check professional summary
   196→  if (!resume.professionalSummary || resume.professionalSummary.trim().length === 0) {
   197→    score -= 20;
   198→    issues.push({
   199→      issue: 'Missing professional summary',
   200→      suggestion: 'Add a 2-3 sentence professional summary highlighting your key achievements and target role',
   201→      severity: 'error',
   202→    });
   203→  } else if (resume.professionalSummary.trim().length <= 20) {
   204→    score -= 12;
   205→    issues.push({
   206→      issue: 'Professional summary is too short',
   207→      suggestion: 'Expand your summary to 2-3 sentences highlighting key achievements and target role',
   208→      severity: 'warning',
   209→    });
   210→  }
   211→
   212→  // Check work experience
   213→  if (!resume.workExperience || resume.workExperience.length === 0) {
   214→    score -= 25;
   215→    issues.push({
   216→      issue: 'No work experience entries',
   217→      suggestion: 'Add at least one work experience entry with job title, company, and description',
   218→      severity: 'error',
   219→    });
   220→  } else {
   221→    const hasDescription = resume.workExperience.some(
   222→      (exp) => exp.description && exp.description.trim().length > 0
   223→    );
   224→    if (!hasDescription) {
   225→      score -= 15;
   226→      issues.push({
   227→        issue: 'Work experience entries lack descriptions',
   228→        suggestion: 'Add bullet-point descriptions for each role highlighting responsibilities and achievements',
   229→        severity: 'error',
   230→      });
   231→    }
   232→  }
   233→
   234→  // Check education
   235→  if (!resume.education || resume.education.length === 0) {
   236→    score -= 15;
   237→    issues.push({
   238→      issue: 'No education entries',
   239→      suggestion: 'Add at least one education entry with degree, field of study, and school',
   240→      severity: 'warning',
   241→    });
   242→  }
   243→
   244→  // Check skills
   245→  if (!resume.skills || resume.skills.length === 0) {
   246→    score -= 15;
   247→    issues.push({
   248→      issue: 'No skills listed',
   249→      suggestion: 'Add at least 3 relevant skills to help ATS systems match your resume',
   250→      severity: 'error',
   251→    });
   252→  } else if (resume.skills.length < 3) {
   253→    score -= 8;
   254→    issues.push({
   255→      issue: 'Too few skills listed',
   256→      suggestion: 'Add more skills — aim for at least 7-10 relevant skills for better ATS matching',
   257→      severity: 'warning',
   258→    });
   259→  }
   260→
   261→  // Check date consistency in work experience
   262→  if (resume.workExperience && resume.workExperience.length > 0) {
   263→    const datePattern = /^(jan|feb|mar|apr|may|jun|jul|aug|sep|oct|nov|dec|january|february|march|april|may|june|july|august|september|october|november|december)\s*\d{4}$/i;
   264→    const datePattern2 = /^\d{1,2}\/\d{1,2}\/\d{2,4}$/;
   265→    const datePattern3 = /^\d{4}$/;
   266→
   267→    for (const exp of resume.workExperience) {
   268→      if (exp.startDate && exp.startDate.trim()) {
   269→        const startMatch = datePattern.test(exp.startDate.trim()) ||
   270→          datePattern2.test(exp.startDate.trim()) ||
   271→          datePattern3.test(exp.startDate.trim());
   272→        if (!startMatch) {
   273→          score -= 3;
   274→          issues.push({
   275→            issue: `Inconsistent date format at "${exp.position}" at ${exp.company}`,
   276→            suggestion: 'Use a consistent date format like "Jan 2023" or "01/2023" throughout',
   277→            severity: 'warning',
   278→          });
   279→          break; // Only flag once
   280→        }
   281→      }
   282→    }
   283→  }
   284→
   285→  return { score: Math.max(0, Math.min(100, score)), issues };
   286→}

   288→function analyzeWorkExperience(workExperience: WorkExperience[]): { score: number; suggestions: Suggestion[] } {
   289→  if (!workExperience || workExperience.length === 0) {
   290→    return {
   291→      score: 0,
   292→      suggestions: [
   293→        {
   294→          category: 'workExperience',
   295→          message: 'Add work experience to significantly improve your resume score',
   296→          priority: 'high',
   297→          section: 'workExperience',
   298→        },
   299→      ],
   300→    };
   301→  }
   302→
   303→  let score = 0;
   304→  const suggestions: Suggestion[] = [];
   305→
   306→  // Number of entries scoring
   307→  const count = workExperience.length;
   308→  if (count === 1) score = 40;
   309→  else if (count === 2) score = 70;
   310→  else if (count >= 3) score = 90;
   311→
   312→  // Average description length
   313→  const entriesWithDesc = workExperience.filter((e) => e.description && e.description.trim().length > 0);
   314→  if (entriesWithDesc.length > 0) {
   315→    const avgDescLen = entriesWithDesc.reduce((sum, e) => sum + e.description.trim().length, 0) / entriesWithDesc.length;
   316→
   317→    if (avgDescLen < 50) {
   318→      score -= 20;
   319→      suggestions.push({
   320→        category: 'workExperience',
   321→        message: 'Work experience descriptions are too brief — add more detail about your responsibilities and achievements',
   322→        priority: 'high',
   323→        section: 'workExperience',
   324→      });
   325→    } else if (avgDescLen < 150) {
   326→      score -= 10;
   327→      suggestions.push({
   328→        category: 'workExperience',
   329→        message: 'Expand your work experience descriptions with more specific details and accomplishments',
   330→        priority: 'medium',
   331→        section: 'workExperience',
   332→      });
   333→    } else if (avgDescLen >= 150) {
   334→      score += 5; // Bonus for detailed descriptions
   335→    }
   336→  } else {
   337→    score -= 20;
   338→    suggestions.push({
   339→      category: 'workExperience',
   340→      message: 'Add descriptions to your work experience entries with bullet points',
   341→      priority: 'high',
   342→      section: 'workExperience',
   343→    });
   344→  }
   345→
   346→  // Current job flag
   347→  const hasCurrentJob = workExperience.some((e) => e.currentJob);
   348→  if (hasCurrentJob) {
   349→    score += 5;
   350→  }
   351→
   352→  // Action verbs check
   353→  let totalBullets = 0;
   354→  let actionVerbBullets = 0;
   355→
   356→  for (const exp of entriesWithDesc) {
   357→    const bullets = exp.description
   358→      .split('\n')
   359→      .map((b) => b.trim())
   360→      .filter((b) => b.length > 0 && (b.startsWith('-') || b.startsWith('•') || b.startsWith('*')));
   361→
   362→    for (const bullet of bullets) {
   363→      const cleanBullet = bullet.replace(/^[-•*]\s*/, '').trim();
   364→      if (cleanBullet.length === 0) continue;
   365→      totalBullets++;
   366→      const firstWord = cleanBullet.split(/\s+/)[0];
   367→      if (ACTION_VERBS.some((verb) => firstWord.toLowerCase() === verb.toLowerCase())) {
   368→        actionVerbBullets++;
   369→      }
   370→    }
   371→  }
   372→
   373→  if (totalBullets > 0) {
   374→    const actionVerbRatio = actionVerbBullets / totalBullets;
   375→    if (actionVerbRatio < 0.3) {
   376→      score -= 10;
   377→      suggestions.push({
   378→        category: 'workExperience',
   379→        message: 'Start more bullet points with strong action verbs (Led, Developed, Achieved, Implemented, etc.)',
   380→        priority: 'medium',
   381→        section: 'workExperience',
   382→      });
   383→    } else if (actionVerbRatio >= 0.6) {
   384→      score += 5; // Bonus for strong action verbs
   385→    }
   386→  }
   387→
   388→  return { score: Math.max(0, Math.min(100, score)), suggestions };
   389→}

   504→function analyzeReadability(resume: ResumeData): number {
   505→  // Total word count across all text sections
   506→  const texts: string[] = [];
   507→
   508→  if (resume.professionalSummary) texts.push(resume.professionalSummary);
   509→  if (resume.personalInfo?.professionalTitle) texts.push(resume.personalInfo.professionalTitle);
   510→
   511→  if (resume.workExperience) {
   512→    for (const exp of resume.workExperience) {
   513→      if (exp.description) texts.push(exp.description);
   514→      if (exp.position) texts.push(exp.position);
   515→    }
   516→  }
   517→
   518→  if (resume.education) {
   519→    for (const edu of resume.education) {
   520→      if (edu.degree) texts.push(edu.degree);
   521→      if (edu.fieldOfStudy) texts.push(edu.fieldOfStudy);
   522→    }
   523→  }
   524→
   525→  if (resume.projects) {
   526→    for (const proj of resume.projects) {
   527→      if (proj.description) texts.push(proj.description);
   528→    }
   529→  }
   530→
   531→  if (resume.additionalInfo?.text) texts.push(resume.additionalInfo.text);
   532→
   533→  const totalWords = texts.reduce((sum, text) => sum + wordCount(text), 0);
   534→
   535→  // Ideal range: 300-800 words
   536→  if (totalWords < 100) return 20;
   537→  if (totalWords < 200) return 45;
   538→  if (totalWords < 300) return 65;
   539→  if (totalWords <= 800) return 90;
   540→  if (totalWords <= 1000) return 80;
   541→  if (totalWords <= 1200) return 65;
   542→  return 45; // Way too long
   543→}

   545→function analyzeAchievements(workExperience: WorkExperience[]): { score: number; suggestions: Suggestion[] } {
   546→  if (!workExperience || workExperience.length === 0) {
   547→    return { score: 0, suggestions: [] };
   548→  }
   549→
   550→  const suggestions: Suggestion[] = [];
   551→  let totalBullets = 0;
   552→  let quantifiableBullets = 0;
   553→
   554→  const metricsPattern = /\d+%|\$[\d,.]+|\d+\+?\s*(?:users?|clients?|customers?|members?|people|projects?|teams?|reports?|sites?|pages?|articles?|products?|services?|applications?|systems?|processes?|campaigns?|accounts?)/i;
   555→  const numberPattern = /\b\d{2,}\b/;
   556→
   557→  for (const exp of workExperience) {
   558→    if (!exp.description) continue;
   559→    const bullets = exp.description
   560→      .split('\n')
   561→      .map((b) => b.trim())
   562→      .filter((b) => b.length > 0);
   563→
   564→    for (const bullet of bullets) {
   565→      const clean = bullet.replace(/^[-•*]\s*/, '').trim();
   566→      if (clean.length === 0) continue;
   567→      totalBullets++;
   568→
   569→      if (metricsPattern.test(clean) || numberPattern.test(clean)) {
   570→        quantifiableBullets++;
   571→      }
   572→    }
   573→  }
   574→
   575→  if (totalBullets === 0) return { score: 10, suggestions };
   576→
   577→  const ratio = quantifiableBullets / totalBullets;
   578→  let score: number;
   579→
   580→  if (ratio >= 0.6) score = 95;
   581→  else if (ratio >= 0.4) score = 80;
   582→  else if (ratio >= 0.2) score = 60;
   583→  else if (ratio >= 0.1) score = 40;
   584→  else score = 20;
   585→
   586→  if (ratio < 0.3) {
   587→    suggestions.push({
   588→      category: 'achievements',
   589→      message: 'Add quantifiable achievements with numbers, percentages, or dollar amounts to your bullet points',
   590→      priority: 'high',
   591→      section: 'workExperience',
   592→    });
   593→  } else if (ratio < 0.5) {
   594→    suggestions.push({
   595→      category: 'achievements',
   596→      message: 'More of your bullet points should include quantifiable results (e.g., "Increased sales by 25%")',
   597→      priority: 'medium',
   598→      section: 'workExperience',
   599→    });
   600→  }
   601→
   602→  return { score, suggestions };
   603→}

   639→  .join('\n')}
   640→  `.trim();
   641→

   643→
   644→  if (jobDescription && jobDescription.trim()) {
   645→    prompt = `Analyze this resume against the following job description and respond ONLY with valid JSON, no markdown.
   646→
   647→JOB DESCRIPTION:
   648→${jobDescription}
   649→
   650→RESUME:
   651→${resumeText}
   652→
   653→Respond with this exact JSON structure:
   654→{
   655→  "keywordAnalysis": {
   656→    "matched": ["keyword1", "keyword2"],
   657→    "missing": ["keyword1", "keyword2"],

   670→}

   673→- "matched": skills/keywords from the job description found in the resume (case-insensitive)
   674→- "missing": important skills/keywords from the job description NOT found in the resume
   675→- "suggested": additional relevant keywords that could strengthen alignment
   676→- "matchPercentage": percentage of critical job keywords found in resume (0-100)
   677→- "grammarIssues": any grammar, spelling, or readability issues found in the resume text
   678→- "grammarScore": overall grammar and language quality (0-100)
   679→- "keywordScore": how well the resume's keywords align with the job (0-100)
   680→- "skillsMatchScore": how well listed skills match job requirements (0-100)
   681→- "suggestions": 3-5 actionable improvement suggestions
   682→- Only return the JSON, no explanation text`;
   683→  } else {
   684→    prompt = `Analyze this resume for ATS optimization and respond ONLY with valid JSON, no markdown.
   685→
   686→RESUME:
   687→${resumeText}
   688→
   689→Respond with this exact JSON structure:
   690→{
   691→  "keywordAnalysis": {
   692→    "matched": ["existing keyword 1", "existing keyword 2"],
   693→    "missing": [],
   694→    "suggested": ["suggested keyword 1", "suggested keyword 2", "suggested keyword 3"],
   695→    "matchPercentage": 70
   696→  },
   697→  "grammarIssues": [
   698→    { "message": "description of issue", "suggestion": "how to fix it", "severity": "warning" }
   699→  ],
   700→  "grammarScore": 85,
   701→  "keywordScore": 70,
   702→  "skillsMatchScore": 70,
   703→  "suggestions": [
   704→    { "category": "keywords", "message": "actionable suggestion", "priority": "high", "section": "skills" }
   705→  ]
   706→}
   707→
   708→Instructions:
   709→- "matched": strong, relevant keywords already present in the resume
   710→- "missing": empty array (no job description to compare against)
   711→- "suggested": 5-10 industry-relevant keywords/skills the resume should include based on the target role/industry
   712→- "matchPercentage": estimated ATS keyword strength (0-100)
   713→- "grammarIssues": any grammar, spelling, readability, or weak verb usage issues found
   714→- "grammarScore": overall grammar and language quality (0-100)
   715→- "keywordScore": keyword strength and relevance (0-100)
   716→- "skillsMatchScore": how comprehensive the skills section is (0-100)
   717→- "suggestions": 3-5 actionable improvement suggestions covering grammar, keywords, content strength, and ATS optimization
   718→- Only return the JSON, no explanation text`;
   719→  }

   722→    const completion = await zai.chat.completions.create({
   723→      messages: [
   724→        {
   725→          role: 'assistant',
   726→          content:
   727→            'You are an ATS resume analysis expert. Respond ONLY with valid JSON, no markdown.',
   728→        },
   729→        { role: 'user', content: prompt },
   730→      ],
   731→      thinking: { type: 'disabled' },

   756→            ? Math.max(0, Math.min(100, parsed.keywordAnalysis.matchPercentage))
   757→            : 50,
   758→      },
   759→      grammarIssues: Array.isArray(parsed.grammarIssues)
   760→        ? parsed.grammarIssues
   761→            .filter(
   762→              (g: Record<string, unknown>) =>
   763→                g.message && g.suggestion && ['error', 'warning', 'info'].includes(g.severity as string)

   767→      grammarScore:
   768→        typeof parsed.grammarScore === 'number'
   769→          ? Math.max(0, Math.min(100, parsed.grammarScore))
   770→          : 75,
   771→      keywordScore:
   772→        typeof parsed.keywordScore === 'number'
   773→          ? Math.max(0, Math.min(100, parsed.keywordScore))
   774→          : 60,
   775→      skillsMatchScore:
   776→        typeof parsed.skillsMatchScore === 'number'
   777→          ? Math.max(0, Math.min(100, parsed.skillsMatchScore))
   778→          : 60,
   779→      aiSuggestions: Array.isArray(parsed.suggestions)
   780→        ? parsed.suggestions
   781→            .filter(
   782→              (s: Record<string, unknown>) =>
   783→                s.category && s.message && ['high', 'medium', 'low'].includes(s.priority as string)
   784→            )
   785→            .slice(0, 10)
   786→        : [],
   787→    };
   788→  } catch (error) {

   790→    // Return sensible defaults on AI failure
   791→    return {
   792→      keywordAnalysis: {
   793→        matched: (resume.skills || []).map((s) => s.name).slice(0, 10),
   794→        missing: [],
   795→        suggested: [],
   796→        matchPercentage: 50,

   799→      grammarScore: 75,

   807→// ─── Score Combination ──────────────────────────────────────────────────
   808→
   809→function calculateOverallScore(scores: {
   810→  contactInfo: number;
   811→  formatting: number;
   812→  keywords: number;
   813→  skillsMatch: number;
   814→  workExperience: number;
   815→  education: number;
   816→  grammar: number;
   817→  readability: number;
   818→  achievements: number;
   819→}): number {
   820→  return Math.round(
   821→    scores.contactInfo * 0.10 +
   822→      scores.formatting * 0.15 +
   823→      scores.keywords * 0.15 +
   824→      scores.skillsMatch * 0.15 +
   825→      scores.workExperience * 0.15 +
   826→      scores.education * 0.10 +
   827→      scores.grammar * 0.08 +
   828→      scores.readability * 0.07 +
   829→      scores.achievements * 0.05
   830→  );
   831→}

   835→export async function POST(request: NextRequest) {
   836→  try {
   837→    // 1. Parse request body
   838→    let body: { resumeId?: string; jobDescription?: string };
   839→    try {
   840→      body = await request.json();
   841→    } catch {
   842→      return NextResponse.json(
   843→        { success: false, error: 'Invalid request body' },
   844→        { status: 400 }
   845→      );
   846→    }
   847→
   848→    const { resumeId, jobDescription } = body;
   849→
   850→    if (!resumeId || typeof resumeId !== 'string') {
   851→      return NextResponse.json(
   852→        { success: false, error: 'resumeId is required' },
   853→        { status: 400 }
   854→      );
   855→    }
   856→
   857→    // 2. Authentication
   858→    const user = await getAuthUser(request);
   859→    if (!user) {
   860→      return NextResponse.json(
   861→        { success: false, error: 'Unauthorized' },
   862→        { status: 401 }
   863→      );
   864→    }
   865→
   866→    // 3. Feature access check
   867→    const access = await checkAndIncrementUsage(user.userId, FEATURE_KEYS.ATS_CHECKER);
   868→    if (!access.allowed) {
   869→      return NextResponse.json(
   870→        {
   871→          success: false,
   872→          error: 'limit_reached',
   873→          limit: access.limit,
   874→          used: access.used,
   875→          planName: access.planName,
   876→        },
   877→        { status: 403 }
   878→      );
   879→    }
   880→
   881→    // 4. Fetch resume and verify ownership
   882→    const resume = await db.resume.findUnique({
   883→      where: { id: resumeId },
   884→    });
   885→
   886→    if (!resume) {
   887→      return NextResponse.json(
   888→        { success: false, error: 'Resume not found' },
   889→        { status: 404 }
   890→      );
   891→    }
   892→
   893→    if (resume.user_id !== user.userId) {
   894→      return NextResponse.json(
   895→        { success: false, error: 'Resume not found' },
   896→        { status: 404 }
   897→      );
   898→    }
   899→
   900→    // 5. Parse resume data
   901→    let resumeData: ResumeData;
   902→    try {
   903→      resumeData = typeof resume.data_json === 'string'
   904→        ? JSON.parse(resume.data_json)
   905→        : resume.data_json;
   906→    } catch {
   907→      return NextResponse.json(
   908→        { success: false, error: 'Invalid resume data' },
   909→        { status: 400 }
   910→      );
   911→    }
   912→
   913→    // 6. Run local deterministic analysis
   914→    const contactResult = analyzeContactInfo(resumeData.personalInfo || {} as PersonalInfo);
   915→    const formattingResult = analyzeFormatting(resumeData);
   916→    const workResult = analyzeWorkExperience(resumeData.workExperience || []);
   917→    const educationResult = analyzeEducation(resumeData.education || []);
   918→    const skillsResult = analyzeSkills(resumeData.skills || []);
   919→    const readabilityScore = analyzeReadability(resumeData);
   920→    const achievementsResult = analyzeAchievements(resumeData.workExperience || []);
   921→
   922→    // 7. Run AI analysis (with graceful fallback)
   923→    const aiResult = await analyzeWithAI(resumeData, jobDescription);
   924→
   925→    // 8. Combine scores
   926→    const categoryScores = {
   927→      contactInfo: contactResult.score,
   928→      formatting: formattingResult.score,
   929→      keywords: aiResult.keywordScore,
   930→      skillsMatch: aiResult.skillsMatchScore,
   931→      workExperience: workResult.score,
   932→      education: educationResult.score,
   933→      grammar: aiResult.grammarScore,
   934→      readability: readabilityScore,
   935→      achievements: achievementsResult.score,
   936→    };
   937→
   938→    const overallScore = calculateOverallScore(categoryScores);
   939→
   940→    // 9. Merge all suggestions
   941→    const allSuggestions: Suggestion[] = [
   942→      ...contactResult.suggestions,
   943→      ...workResult.suggestions,
   944→      ...educationResult.suggestions,
   945→      ...skillsResult.suggestions,
   946→      ...achievementsResult.suggestions,
   947→      ...aiResult.aiSuggestions,
   948→    ];
   949→
   950→    // Sort by priority
   951→    const priorityOrder: Record<string, number> = { high: 0, medium: 1, low: 2 };
   952→    allSuggestions.sort((a, b) => (priorityOrder[a.priority] ?? 2) - (priorityOrder[b.priority] ?? 2));
   953→
   954→    // 10. Build formatted response
   955→    const categoryScoresFormatted: Record<string, CategoryScore> = {
   956→      contactInfo: { score: categoryScores.contactInfo, label: 'Contact Information' },
   957→      formatting: { score: categoryScores.formatting, label: 'Formatting' },
   958→      keywords: { score: categoryScores.keywords, label: 'Keywords' },
   959→      skillsMatch: { score: categoryScores.skillsMatch, label: 'Skills Match' },
   960→      workExperience: { score: categoryScores.workExperience, label: 'Work Experience' },
   961→      education: { score: categoryScores.education, label: 'Education' },
   962→      grammar: { score: categoryScores.grammar, label: 'Grammar' },
   963→      readability: { score: categoryScores.readability, label: 'Readability' },
   964→      achievements: { score: categoryScores.achievements, label: 'Achievements' },
   965→    };
   966→
   967→    // 11. Store in database
   968→    const report = await db.atsReport.create({
   969→      data: {
   970→        user_id: user.userId,
   971→        resume_id: resumeId,
   972→        overall_score: overallScore,
   973→        category_scores: JSON.stringify(categoryScoresFormatted),
   974→        keyword_analysis: JSON.stringify(aiResult.keywordAnalysis),
   975→        grammar_issues: JSON.stringify(aiResult.grammarIssues),
   976→        formatting_issues: JSON.stringify(formattingResult.issues),
   977→        suggestions: JSON.stringify(allSuggestions.slice(0, 20)),
   978→        job_description: jobDescription || '',
   979→        resume_snapshot: resume.data_json,
   980→      },
   981→    });
   982→
   983→    // 12. Return response
   984→    return NextResponse.json({
   985→      success: true,
   986→      report: {
   987→        id: report.id,
   988→        overallScore,
   989→        categoryScores: categoryScoresFormatted,
   990→        keywordAnalysis: aiResult.keywordAnalysis,
   991→        grammarIssues: aiResult.grammarIssues,
   992→        formattingIssues: formattingResult.issues,
   993→        suggestions: allSuggestions.slice(0, 20),
   994→        createdAt: report.created_at.toISOString(),
   995→      },
   996→    });
   997→  } catch (error) {
   998→    console.error('[ATS Check] Unexpected error:', error);
   999→    return NextResponse.json(
  1000→      { success: false, error: 'Internal server error' },
  1001→      { status: 500 }
  1002→    );
  1003→  }
