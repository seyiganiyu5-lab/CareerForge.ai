   109→   109→function formatPrice(cents: number, currency: string = 'USD'): string {
   110→   110→  const amount = cents / 100;
   111→   111→  return new Intl.NumberFormat('en-US', {
   112→   112→    style: 'currency',
   113→   113→    currency,
   114→   114→    minimumFractionDigits: 2,
   115→   115→    maximumFractionDigits: 2,
   116→   116→  }).format(amount);
   117→   117→}
   118→   118→
   119→   119→function formatDate(dateStr: string | null): string {
   120→   120→  if (!dateStr) return 'N/A';
   121→   121→  return new Date(dateStr).toLocaleDateString('en-US', {
   122→   122→    year: 'numeric',
   123→   123→    month: 'long',
   124→   124→    day: 'numeric',
   125→   125→  });
   126→   126→}
   127→   127→
   128→   128→function formatShortDate(dateStr: string | null): string {
   129→   129→  if (!dateStr) return '—';
   130→   130→  return new Date(dateStr).toLocaleDateString('en-US', {
   131→   131→    year: 'numeric',
   132→   132→    month: 'short',
   133→   133→    day: 'numeric',
   134→   134→  });
   135→   135→}

   141→   141→function getFeatureLimitText(feature: PlanFeature): string {
   142→   142→  if (!feature.enabled) return '';
   143→   143→  if (feature.limit === null || feature.limit === -1 || feature.limit === 0) return 'Unlimited';
   144→   144→  return String(feature.limit);

   268→   268→      setLoading(false);

   290→   290→          window.location.href = data.payment_link;
   291→   291→        } else {
   292→   292→          toast.success(`Upgraded to ${planDisplayName}!`);
   293→   293→          await fetchAllData();
   294→   294→        }
   295→   295→      } else {
   296→   296→        toast.error(data.error || 'Upgrade failed');
   297→   297→      }
   298→   298→    } catch {
   299→   299→      toast.error('Something went wrong. Please try again.');
   300→   300→    } finally {
   301→   301→      setUpgrading(null);
   302→   302→    }
   303→   303→  };

   312→   312→        body: JSON.stringify({ immediate: true }),
   313→   313→      });

   317→   317→        toast.success('Successfully downgraded to Free plan');
   318→   318→        setCancelDialogOpen(false);
   319→   319→        await fetchAllData();
   320→   320→      } else {
   321→   321→        toast.error(data.error || 'Failed to cancel');
   322→   322→      }
   323→   323→    } catch {
   324→   324→      toast.error('Something went wrong. Please try again.');
   325→   325→    } finally {
   326→   326→      setCancelling(false);
   327→   327→    }
   328→   328→  };
   329→   329→

   333→   333→    return (
   334→   334→      <div className="space-y-8">
   335→   335→        <div>
   336→   336→          <Skeleton className="h-8 w-48" />

   338→   338→        </div>

   341→   341→          <SkeletonCard />
   342→   342→          <SkeletonCard />
   343→   343→          <SkeletonCard />
   344→   344→        </div>
   345→   345→        <Skeleton className="h-60 w-full rounded-xl" />
   346→   346→      </div>
   347→   347→    );
   348→   348→  }
   349→   349→
   350→   350→  // Collect all feature keys from all plans for the comparison table
   351→   351→  const allFeatureKeys = Array.from(
   352→   352→    new Set(plans.flatMap(p => p.features.map(f => f.key)))
   353→   353→  ).filter(key => FEATURE_DISPLAY_INFO[key]);
   354→   354→
   355→   355→  return (
   356→   356→    <motion.div
   357→   357→      variants={staggerContainer}
   358→   358→      initial="hidden"
   359→   359→      animate="visible"
   360→   360→      className="space-y-8"
   361→   361→    >
   362→   362→      {/* Page Header */}
   363→   363→      <motion.div variants={fadeInUp}>
   364→   364→        <h1 className="text-2xl md:text-3xl font-bold font-[family-name:var(--font-poppins)] text-foreground">
   365→   365→          Billing & Plans

   368→   368→          Manage your subscription, view usage, and billing history.
   369→   369→        </p>

   372→   372→      {/* ── 1. Current Plan Section ────────────────────────────────────────── */}

   374→   374→        <Card className="overflow-hidden">
