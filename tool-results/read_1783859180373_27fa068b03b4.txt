    86→    86→const TABLET_BREAKPOINT = 1024;
    87→    87→
    88→    88→function useIsTablet() {
    89→    89→  const [isTablet, setIsTablet] = useState(false);
    90→    90→  useEffect(() => {
    91→    91→    const update = () => {
    92→    92→      const w = window.innerWidth;
    93→    93→      setIsTablet(w >= 768 && w < TABLET_BREAKPOINT);
    94→    94→    };
    95→    95→    const mql = window.matchMedia(`(min-width: 768px) and (max-width: ${TABLET_BREAKPOINT - 1}px)`);
    96→    96→    mql.addEventListener('change', () => update());
    97→    97→    update();
    98→    98→    return () => mql.removeEventListener('change', () => update());
    99→    99→  }, []);
   100→   100→  return isTablet;
   101→   101→}

   426→   426→  const handleAiApply = useCallback((text: string) => {
