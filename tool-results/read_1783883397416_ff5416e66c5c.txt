   108→function analyzeContactInfo(personalInfo: PersonalInfo): { score: number; suggestions: Suggestion[] } {

   111→
   112→  const hasEmail = !!personalInfo.email && personalInfo.email.trim().length > 0;

   119→  const hasTitle = !!personalInfo.professionalTitle && personalInfo.professionalTitle.trim().length > 0;

   136→      priority: 'high',
   137→      section: 'personalInfo',
   138→    });

   142→    suggestions.push({
   143→      category: 'contactInfo',
   144→      message: 'Add your location (city, state) — many employers filter by location',
   145→      priority: 'high',
   146→      section: 'personalInfo',
   147→    });

   191→function analyzeFormatting(resume: ResumeData): { score: number; issues: FormattingIssue[] } {
   192→  let score = 100;
   193→  const issues: FormattingIssue[] = [];
   194→
   195→  // Check professional summary
   196→  if (!resume.professionalSummary || resume.professionalSummary.trim().length === 0) {
   197→    score -= 20;
   198→    issues.push({
   199→      issue: 'Missing professional summary',
   200→      suggestion: 'Add a 2-3 sentence professional summary highlighting your key achievements and target role',
   201→      severity: 'error',
   202→    });
   203→  } else if (resume.professionalSummary.trim().length <= 20) {
   204→    score -= 12;
   205→    issues.push({
   206→      issue: 'Professional summary is too short',
   207→      suggestion: 'Expand your summary to 2-3 sentences highlighting key achievements and target role',
   208→      severity: 'warning',
   209→    });
   210→  }
   211→
   212→  // Check work experience
   213→  if (!resume.workExperience || resume.workExperience.length === 0) {
   214→    score -= 25;
   215→    issues.push({
   216→      issue: 'No work experience entries',
   217→      suggestion: 'Add at least one work experience entry with job title, company, and description',
   218→      severity: 'error',
   219→    });
   220→  } else {
   221→    const hasDescription = resume.workExperience.some(
   222→      (exp) => exp.description && exp.description.trim().length > 0
   223→    );
   224→    if (!hasDescription) {
   225→      score -= 15;
   226→      issues.push({
   227→        issue: 'Work experience entries lack descriptions',
   228→        suggestion: 'Add bullet-point descriptions for each role highlighting responsibilities and achievements',
   229→        severity: 'error',
   230→      });
   231→    }
   232→  }
   233→
   234→  // Check education
   235→  if (!resume.education || resume.education.length === 0) {
   236→    score -= 15;
   237→    issues.push({
   238→      issue: 'No education entries',
   239→      suggestion: 'Add at least one education entry with degree, field of study, and school',
   240→      severity: 'warning',
   241→    });
   242→  }
   243→
   244→  // Check skills
   245→  if (!resume.skills || resume.skills.length === 0) {
   246→    score -= 15;
   247→    issues.push({
   248→      issue: 'No skills listed',
   249→      suggestion: 'Add at least 3 relevant skills to help ATS systems match your resume',
   250→      severity: 'error',
   251→    });
   252→  } else if (resume.skills.length < 3) {
   253→    score -= 8;
   254→    issues.push({
   255→      issue: 'Too few skills listed',
   256→      suggestion: 'Add more skills — aim for at least 7-10 relevant skills for better ATS matching',
   257→      severity: 'warning',
   258→    });
   259→  }
   260→
   261→  // Check date consistency in work experience
   262→  if (resume.workExperience && resume.workExperience.length > 0) {
   263→    // Recognized date formats: "Jan 2023", "January 2023", "01/2023", "01/01/2023", "2023", "2024-12" (YYYY-MM)
   264→    const datePatterns = [
   265→      /^(jan|feb|mar|apr|may|jun|jul|aug|sep|oct|nov|dec|january|february|march|april|may|june|july|august|september|october|november|december)\s*\d{4}$/i,
   266→      /^\d{1,2}\/\d{4}$/,
   267→      /^\d{1,2}\/\d{1,2}\/\d{2,4}$/,
   268→      /^\d{4}$/,
   269→      /^\d{4}-\d{1,2}$/,  // YYYY-MM format (e.g., 2024-12)
   270→    ];
   271→
   272→    // First, determine which format is used by the first entry
   273→    let detectedFormat: number | null = null;
   274→    const dates = resume.workExperience
   275→      .flatMap((exp) => [exp.startDate, exp.endDate, exp.current ? 'Present' : null])
   276→      .filter((d): d is string => !!d && d.trim() !== '' && d.trim().toLowerCase() !== 'present');
   277→
   278→    for (const date of dates) {
   279→      for (let i = 0; i < datePatterns.length; i++) {
   280→        if (datePatterns[i].test(date.trim())) {
   281→          if (detectedFormat === null) {
   282→            detectedFormat = i;
   283→          } else if (detectedFormat !== i) {
   284→            // Inconsistent: different format detected
   285→            score -= 3;
   286→            issues.push({
   287→              issue: 'Mixed date formats detected across work experience entries',
   288→              suggestion: 'Use a consistent date format like "Jan 2023", "01/2023", or "2024-12" throughout',
   289→              severity: 'warning',
   290→            });
   291→            break;
   292→          }
   293→          break;
   294→        }
   295→      }
   296→      // If a date doesn't match ANY known format, flag it
   297→      if (detectedFormat !== null) break; // already flagged inconsistency
   298→      const matchesAny = datePatterns.some((p) => p.test(date.trim()));
   299→      if (!matchesAny) {
   300→        score -= 3;
   301→        issues.push({
   302→          issue: `Unrecognized date format: "${date}"`,
   303→          suggestion: 'Use a standard format like "Jan 2023", "MM/YYYY", or "YYYY-MM"',
   304→          severity: 'warning',
   305→        });
   306→        break;
   307→      }
   308→    }
   309→  }
   310→
   311→  return { score: Math.max(0, Math.min(100, score)), issues };
   312→}

   314→function analyzeWorkExperience(workExperience: WorkExperience[]): { score: number; suggestions: Suggestion[] } {
   315→  if (!workExperience || workExperience.length === 0) {
   316→    return {
   317→      score: 0,
   318→      suggestions: [
   319→        {
   320→          category: 'workExperience',
   321→          message: 'Add work experience to significantly improve your resume score',
   322→          priority: 'high',
   323→          section: 'workExperience',
   324→        },
   325→      ],
   326→    };
   327→  }
   328→
   329→  let score = 0;
   330→  const suggestions: Suggestion[] = [];
   331→
   332→  // Number of entries scoring
   333→  const count = workExperience.length;
   334→  if (count === 1) score = 40;
   335→  else if (count === 2) score = 70;
   336→  else if (count >= 3) score = 90;
   337→
   338→  // Average description length
   339→  const entriesWithDesc = workExperience.filter((e) => e.description && e.description.trim().length > 0);
   340→  if (entriesWithDesc.length > 0) {
   341→    const avgDescLen = entriesWithDesc.reduce((sum, e) => sum + e.description.trim().length, 0) / entriesWithDesc.length;
   342→
   343→    if (avgDescLen < 50) {
   344→      score -= 20;
   345→      suggestions.push({
   346→        category: 'workExperience',
   347→        message: 'Work experience descriptions are too brief — add more detail about your responsibilities and achievements',
   348→        priority: 'high',
   349→        section: 'workExperience',
   350→      });
   351→    } else if (avgDescLen < 150) {
   352→      score -= 10;
   353→      suggestions.push({
   354→        category: 'workExperience',
   355→        message: 'Expand your work experience descriptions with more specific details and accomplishments',
   356→        priority: 'medium',
   357→        section: 'workExperience',
   358→      });
   359→    } else if (avgDescLen >= 150) {
   360→      score += 5; // Bonus for detailed descriptions
   361→    }
   362→  } else {
   363→    score -= 20;
   364→    suggestions.push({
   365→      category: 'workExperience',
   366→      message: 'Add descriptions to your work experience entries with bullet points',
   367→      priority: 'high',
   368→      section: 'workExperience',
   369→    });
   370→  }
   371→
   372→  // Current job flag
   373→  const hasCurrentJob = workExperience.some((e) => e.currentJob);
   374→  if (hasCurrentJob) {
   375→    score += 5;
   376→  }
   377→
   378→  // Action verbs check
   379→  let totalBullets = 0;
   380→  let actionVerbBullets = 0;
   381→
   382→  for (const exp of entriesWithDesc) {
   383→    const bullets = exp.description
   384→      .split('\n')
   385→      .map((b) => b.trim())
   386→      .filter((b) => b.length > 0 && (b.startsWith('-') || b.startsWith('•') || b.startsWith('*')));
   387→
   388→    for (const bullet of bullets) {
   389→      const cleanBullet = bullet.replace(/^[-•*]\s*/, '').trim();
   390→      if (cleanBullet.length === 0) continue;
   391→      totalBullets++;
   392→      const firstWord = cleanBullet.split(/\s+/)[0];
   393→      if (ACTION_VERBS.some((verb) => firstWord.toLowerCase() === verb.toLowerCase())) {
   394→        actionVerbBullets++;
   395→      }
   396→    }
   397→  }
   398→
   399→  if (totalBullets > 0) {
   400→    const actionVerbRatio = actionVerbBullets / totalBullets;
   401→    if (actionVerbRatio < 0.3) {
   402→      score -= 10;
   403→      suggestions.push({
   404→        category: 'workExperience',
   405→        message: 'Start more bullet points with strong action verbs (Led, Developed, Achieved, Implemented, etc.)',
   406→        priority: 'medium',
   407→        section: 'workExperience',
   408→      });
   409→    } else if (actionVerbRatio >= 0.6) {
   410→      score += 5; // Bonus for strong action verbs
   411→    }
   412→  }
   413→
   414→  return { score: Math.max(0, Math.min(100, score)), suggestions };
   415→}

   530→function analyzeReadability(resume: ResumeData): number {
   531→  // Total word count across all text sections
   532→  const texts: string[] = [];
   533→
   534→  if (resume.professionalSummary) texts.push(resume.professionalSummary);
   535→  if (resume.personalInfo?.professionalTitle) texts.push(resume.personalInfo.professionalTitle);
   536→
   537→  if (resume.workExperience) {
   538→    for (const exp of resume.workExperience) {
   539→      if (exp.description) texts.push(exp.description);
   540→      if (exp.position) texts.push(exp.position);
   541→    }
   542→  }
   543→
   544→  if (resume.education) {
   545→    for (const edu of resume.education) {
   546→      if (edu.degree) texts.push(edu.degree);
   547→      if (edu.fieldOfStudy) texts.push(edu.fieldOfStudy);
   548→    }
   549→  }
   550→
   551→  if (resume.projects) {
   552→    for (const proj of resume.projects) {
   553→      if (proj.description) texts.push(proj.description);
   554→    }
   555→  }
   556→
   557→  if (resume.additionalInfo?.text) texts.push(resume.additionalInfo.text);
   558→
   559→  const totalWords = texts.reduce((sum, text) => sum + wordCount(text), 0);
   560→
   561→  // Ideal range: 300-800 words
   562→  if (totalWords < 100) return 20;
   563→  if (totalWords < 200) return 45;
   564→  if (totalWords < 300) return 65;
   565→  if (totalWords <= 800) return 90;
   566→  if (totalWords <= 1000) return 80;
   567→  if (totalWords <= 1200) return 65;
   568→  return 45; // Way too long
   569→}

   571→function analyzeAchievements(workExperience: WorkExperience[]): { score: number; suggestions: Suggestion[] } {
   572→  if (!workExperience || workExperience.length === 0) {
   573→    return { score: 0, suggestions: [] };
   574→  }
   575→
   576→  const suggestions: Suggestion[] = [];
   577→  let totalBullets = 0;
   578→  let quantifiableBullets = 0;
   579→
   580→  const metricsPattern = /\d+%|\$[\d,.]+|\d+\+?\s*(?:users?|clients?|customers?|members?|people|projects?|teams?|reports?|sites?|pages?|articles?|products?|services?|applications?|systems?|processes?|campaigns?|accounts?)/i;
   581→  const numberPattern = /\b\d{2,}\b/;
   582→
   583→  for (const exp of workExperience) {
   584→    if (!exp.description) continue;
   585→    const bullets = exp.description
   586→      .split('\n')
   587→      .map((b) => b.trim())
   588→      .filter((b) => b.length > 0);
   589→
   590→    for (const bullet of bullets) {
   591→      const clean = bullet.replace(/^[-•*]\s*/, '').trim();
   592→      if (clean.length === 0) continue;
   593→      totalBullets++;
   594→
   595→      if (metricsPattern.test(clean) || numberPattern.test(clean)) {
   596→        quantifiableBullets++;
   597→      }
   598→    }
   599→  }
   600→
   601→  if (totalBullets === 0) return { score: 10, suggestions };
   602→
   603→  const ratio = quantifiableBullets / totalBullets;
   604→  let score: number;
   605→
   606→  if (ratio >= 0.6) score = 95;
   607→  else if (ratio >= 0.4) score = 80;
   608→  else if (ratio >= 0.2) score = 60;
   609→  else if (ratio >= 0.1) score = 40;
   610→  else score = 20;
   611→
   612→  if (ratio < 0.3) {
   613→    suggestions.push({
   614→      category: 'achievements',
   615→      message: 'Add quantifiable achievements with numbers, percentages, or dollar amounts to your bullet points',
   616→      priority: 'high',
   617→      section: 'workExperience',
   618→    });
   619→  } else if (ratio < 0.5) {
   620→    suggestions.push({
   621→      category: 'achievements',
   622→      message: 'More of your bullet points should include quantifiable results (e.g., "Increased sales by 25%")',
   623→      priority: 'medium',
   624→      section: 'workExperience',
   625→    });
   626→  }
   627→
   628→  return { score, suggestions };
   629→}

   673→JOB DESCRIPTION:
   674→${jobDescription}
   675→
   676→RESUME:
   677→${resumeText}

   679→Respond with this exact JSON structure:

   701→- "suggested": additional relevant keywords that could strengthen alignment
   702→- "matchPercentage": percentage of critical job keywords found in resume (0-100)
   703→- "grammarIssues": any grammar, spelling, or readability issues found in the resume text. Do NOT flag date format issues — dates in formats like "Jan 2023", "MM/YYYY", "YYYY-MM", or "YYYY" are all valid.
   704→- "grammarScore": overall grammar and language quality (0-100)
   705→- "keywordScore": how well the resume's keywords align with the job (0-100)

   711→
   712→RESUME:
   713→${resumeText}
   714→
   715→Respond with this exact JSON structure:
   716→{
   717→  "keywordAnalysis": {
   718→    "matched": ["existing keyword 1", "existing keyword 2"],
   719→    "missing": [],
   720→    "suggested": ["suggested keyword 1", "suggested keyword 2", "suggested keyword 3"],
   721→    "matchPercentage": 70
   722→  },
   723→  "grammarIssues": [
   724→    { "message": "description of issue", "suggestion": "how to fix it", "severity": "warning" }
   725→  ],
   726→  "grammarScore": 85,
   727→  "keywordScore": 70,
   728→  "skillsMatchScore": 70,

   735→- "matched": strong, relevant keywords already present in the resume
   736→- "missing": empty array (no job description to compare against)
   737→- "suggested": 5-10 industry-relevant keywords/skills the resume should include based on the target role/industry
   738→- "matchPercentage": estimated ATS keyword strength (0-100)
   739→- "grammarIssues": any grammar, spelling, readability, or weak verb usage issues found. Do NOT flag date format issues — dates in formats like "Jan 2023", "MM/YYYY", "YYYY-MM", or "YYYY" are all valid.
   740→- "grammarScore": overall grammar and language quality (0-100)
   741→- "keywordScore": keyword strength and relevance (0-100)
   742→- "skillsMatchScore": how comprehensive the skills section is (0-100)

   750→        {

   782→            ? Math.max(0, Math.min(100, parsed.keywordAnalysis.matchPercentage))
   783→            : 50,
   784→      },
   785→      grammarIssues: Array.isArray(parsed.grammarIssues)
   786→        ? parsed.grammarIssues
   787→            .filter(

   796→          : 75,
   797→      keywordScore:
   798→        typeof parsed.keywordScore === 'number'
   799→          ? Math.max(0, Math.min(100, parsed.keywordScore))
   800→          : 60,
   801→      skillsMatchScore:
   802→        typeof parsed.skillsMatchScore === 'number'
   803→          ? Math.max(0, Math.min(100, parsed.skillsMatchScore))
   804→          : 60,
   805→      aiSuggestions: Array.isArray(parsed.suggestions)
   806→        ? parsed.suggestions
   807→            .filter(
   808→              (s: Record<string, unknown>) =>
   809→                s.category && s.message && ['high', 'medium', 'low'].includes(s.priority as string)

   811→            .slice(0, 10)

   825→      grammarScore: 75,

   833→// ─── Score Combination ──────────────────────────────────────────────────
   834→
   835→function calculateOverallScore(scores: {
   836→  contactInfo: number;
   837→  formatting: number;
   838→  keywords: number;
   839→  skillsMatch: number;
   840→  workExperience: number;
   841→  education: number;
   842→  grammar: number;
   843→  readability: number;
   844→  achievements: number;
   845→}): number {
   846→  return Math.round(
   847→    scores.contactInfo * 0.10 +
   848→      scores.formatting * 0.15 +
   849→      scores.keywords * 0.15 +
   850→      scores.skillsMatch * 0.15 +
   851→      scores.workExperience * 0.15 +
   852→      scores.education * 0.10 +
   853→      scores.grammar * 0.08 +
   854→      scores.readability * 0.07 +
   855→      scores.achievements * 0.05
   856→  );
   857→}

   861→export async function POST(request: NextRequest) {
   862→  try {
   863→    // 1. Parse request body
   864→    let body: { resumeId?: string; jobDescription?: string };
   865→    try {
   866→      body = await request.json();
   867→    } catch {
   868→      return NextResponse.json(
   869→        { success: false, error: 'Invalid request body' },
   870→        { status: 400 }
   871→      );
   872→    }
   873→
   874→    const { resumeId, jobDescription } = body;
   875→
   876→    if (!resumeId || typeof resumeId !== 'string') {
   877→      return NextResponse.json(
   878→        { success: false, error: 'resumeId is required' },
   879→        { status: 400 }
   880→      );
   881→    }
   882→
   883→    // 2. Authentication
   884→    const user = await getAuthUser(request);
   885→    if (!user) {
   886→      return NextResponse.json(
   887→        { success: false, error: 'Unauthorized' },
   888→        { status: 401 }
   889→      );
   890→    }
   891→
   892→    // 3. Feature access check
   893→    const access = await checkAndIncrementUsage(user.userId, FEATURE_KEYS.ATS_CHECKER);
   894→    if (!access.allowed) {
   895→      return NextResponse.json(
   896→        {
   897→          success: false,
   898→          error: 'limit_reached',
   899→          limit: access.limit,
   900→          used: access.used,
   901→          planName: access.planName,
   902→        },
   903→        { status: 403 }
   904→      );
   905→    }
   906→
   907→    // 4. Fetch resume and verify ownership
   908→    const resume = await db.resume.findUnique({
   909→      where: { id: resumeId },
   910→    });
   911→
   912→    if (!resume) {
   913→      return NextResponse.json(
   914→        { success: false, error: 'Resume not found' },
   915→        { status: 404 }
   916→      );
   917→    }
   918→
   919→    if (resume.user_id !== user.userId) {
   920→      return NextResponse.json(
   921→        { success: false, error: 'Resume not found' },
   922→        { status: 404 }
   923→      );
   924→    }
   925→
   926→    // 5. Parse resume data
   927→    let resumeData: ResumeData;
   928→    try {
   929→      resumeData = typeof resume.data_json === 'string'
   930→        ? JSON.parse(resume.data_json)
   931→        : resume.data_json;
   932→      // Migrate old skills without category
   933→      if (resumeData.skills) {
   934→        resumeData.skills = resumeData.skills.map((s: Record<string, string>) => ({
   935→          ...s,
   936→          category: s.category || 'technical',
   937→        }));
   938→      }
   939→    } catch {
   940→      return NextResponse.json(
   941→        { success: false, error: 'Invalid resume data' },
   942→        { status: 400 }
   943→      );
   944→    }
   945→
   946→    // 6. Run local deterministic analysis
   947→    const contactResult = analyzeContactInfo(resumeData.personalInfo || {} as PersonalInfo);
   948→    const formattingResult = analyzeFormatting(resumeData);
   949→    const workResult = analyzeWorkExperience(resumeData.workExperience || []);
   950→    const educationResult = analyzeEducation(resumeData.education || []);
   951→    const skillsResult = analyzeSkills(resumeData.skills || []);
   952→    const readabilityScore = analyzeReadability(resumeData);
   953→    const achievementsResult = analyzeAchievements(resumeData.workExperience || []);
   954→
   955→    // 7. Run AI analysis (with graceful fallback)
   956→    const aiResult = await analyzeWithAI(resumeData, jobDescription);
   957→
   958→    // 8. Combine scores
   959→    const categoryScores = {
   960→      contactInfo: contactResult.score,
   961→      formatting: formattingResult.score,
   962→      keywords: aiResult.keywordScore,
   963→      skillsMatch: aiResult.skillsMatchScore,
   964→      workExperience: workResult.score,
   965→      education: educationResult.score,
   966→      grammar: aiResult.grammarScore,
   967→      readability: readabilityScore,
   968→      achievements: achievementsResult.score,
   969→    };
   970→
   971→    const overallScore = calculateOverallScore(categoryScores);
   972→
   973→    // 9. Merge all suggestions
   974→    const allSuggestions: Suggestion[] = [
   975→      ...contactResult.suggestions,
   976→      ...workResult.suggestions,
   977→      ...educationResult.suggestions,
   978→      ...skillsResult.suggestions,
   979→      ...achievementsResult.suggestions,
   980→      ...aiResult.aiSuggestions,
   981→    ];
   982→
   983→    // Sort by priority
   984→    const priorityOrder: Record<string, number> = { high: 0, medium: 1, low: 2 };
   985→    allSuggestions.sort((a, b) => (priorityOrder[a.priority] ?? 2) - (priorityOrder[b.priority] ?? 2));
   986→
   987→    // 10. Build formatted response
   988→    const categoryScoresFormatted: Record<string, CategoryScore> = {
   989→      contactInfo: { score: categoryScores.contactInfo, label: 'Contact Information' },
   990→      formatting: { score: categoryScores.formatting, label: 'Formatting' },
   991→      keywords: { score: categoryScores.keywords, label: 'Keywords' },
   992→      skillsMatch: { score: categoryScores.skillsMatch, label: 'Skills Match' },
   993→      workExperience: { score: categoryScores.workExperience, label: 'Work Experience' },
   994→      education: { score: categoryScores.education, label: 'Education' },
   995→      grammar: { score: categoryScores.grammar, label: 'Grammar' },
   996→      readability: { score: categoryScores.readability, label: 'Readability' },
   997→      achievements: { score: categoryScores.achievements, label: 'Achievements' },
   998→    };
   999→
  1000→    // 11. Store in database
  1001→    const report = await db.atsReport.create({
  1002→      data: {
  1003→        user_id: user.userId,
  1004→        resume_id: resumeId,
  1005→        overall_score: overallScore,
  1006→        category_scores: JSON.stringify(categoryScoresFormatted),
  1007→        keyword_analysis: JSON.stringify(aiResult.keywordAnalysis),
  1008→        grammar_issues: JSON.stringify(aiResult.grammarIssues),
  1009→        formatting_issues: JSON.stringify(formattingResult.issues),
  1010→        suggestions: JSON.stringify(allSuggestions.slice(0, 20)),
  1011→        job_description: jobDescription || '',
  1012→        resume_snapshot: resume.data_json,
  1013→      },
  1014→    });
  1015→
  1016→    // 12. Return response
  1017→    return NextResponse.json({
  1018→      success: true,
  1019→      report: {
  1020→        id: report.id,
  1021→        overallScore,
  1022→        categoryScores: categoryScoresFormatted,
  1023→        keywordAnalysis: aiResult.keywordAnalysis,
  1024→        grammarIssues: aiResult.grammarIssues,
  1025→        formattingIssues: formattingResult.issues,
  1026→        suggestions: allSuggestions.slice(0, 20),
  1027→        createdAt: report.created_at.toISOString(),
  1028→      },
  1029→    });
  1030→  } catch (error) {
  1031→    console.error('[ATS Check] Unexpected error:', error);
  1032→    return NextResponse.json(
  1033→      { success: false, error: 'Internal server error' },
  1034→      { status: 500 }
  1035→    );
  1036→  }
