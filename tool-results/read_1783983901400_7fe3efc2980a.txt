    78→// ─── Score Ring Component ─────────────────────────────────────────
    79→
    80→function ScoreRing({ score, size = 160 }: { score: number; size?: number }) {
    81→  const radius = (size - 16) / 2;
    82→  const circumference = 2 * Math.PI * radius;
    83→  const offset = circumference - (score / 100) * circumference;
    84→  const color = score >= 80 ? 'text-emerald-500' : score >= 60 ? 'text-amber-500' : score >= 40 ? 'text-orange-500' : 'text-red-500';
    85→  const bgColor = score >= 80 ? 'bg-emerald-500/10' : score >= 60 ? 'bg-amber-500/10' : score >= 40 ? 'bg-orange-500/10' : 'bg-red-500/10';
    86→  const strokeColor = score >= 80 ? '#10b981' : score >= 60 ? '#f59e0b' : score >= 40 ? '#f97316' : '#ef4444';
    87→  const label = score >= 80 ? 'Excellent' : score >= 60 ? 'Good' : score >= 40 ? 'Fair' : 'Needs Work';
    88→
    89→  return (
    90→    <div className="relative flex items-center justify-center">
    91→      <svg width={size} height={size} className="-rotate-90">
    92→        <circle cx={size / 2} cy={size / 2} r={radius} fill="none" stroke="currentColor" strokeWidth="10" className="text-muted/20" />
    93→        <motion.circle
    94→          cx={size / 2} cy={size / 2} r={radius} fill="none" stroke={strokeColor} strokeWidth="10"
    95→          strokeLinecap="round" strokeDasharray={circumference} initial={{ strokeDashoffset: circumference }}
    96→          animate={{ strokeDashoffset: offset }} transition={{ duration: 1.2, ease: 'easeOut', delay: 0.3 }}
    97→        />
    98→      </svg>
    99→      <div className="absolute flex flex-col items-center">
   100→        <motion.span
   101→          className={cn('text-4xl font-bold font-[family-name:var(--font-poppins)]', color)}
   102→          initial={{ opacity: 0, scale: 0.5 }} animate={{ opacity: 1, scale: 1 }}
   103→          transition={{ duration: 0.5, delay: 0.6 }}
   104→        >
   105→          {score}
   106→        </motion.span>
   107→        <span className="text-xs text-muted-foreground font-[family-name:var(--font-inter)]">/ 100</span>
   108→        <Badge variant="secondary" className={cn('mt-1 text-[10px] font-medium', bgColor, color)}>
   109→          {label}
   110→        </Badge>
   111→      </div>
   112→    </div>
   113→  );
   114→}

   116→// ─── Category Bar Component ───────────────────────────────────────
   117→
   118→function CategoryBar({ category }: { category: CategoryResult }) {
   119→  const pct = Math.round((category.score / category.maxScore) * 100);
   120→  const color = pct >= 80 ? 'bg-emerald-500' : pct >= 60 ? 'bg-amber-500' : pct >= 40 ? 'bg-orange-500' : 'bg-red-500';
   121→
   122→  return (
   123→    <div className="space-y-2">
   124→      <div className="flex items-center justify-between">
   125→        <span className="text-sm font-medium font-[family-name:var(--font-inter)] text-foreground">{category.label}</span>
   126→        <span className="text-sm font-semibold font-[family-name:var(--font-inter)] text-muted-foreground">
   127→          {category.score}/{category.maxScore}
   128→        </span>
   129→      </div>
   130→      <div className="h-2 bg-muted rounded-full overflow-hidden">
   131→        <motion.div
   132→          className={cn('h-full rounded-full', color)}
   133→          initial={{ width: 0 }} animate={{ width: `${pct}%` }}
   134→          transition={{ duration: 0.8, ease: 'easeOut' }}
   135→        />
   136→      </div>
   137→      {category.issues.length > 0 && (
   138→        <ul className="space-y-1 mt-2">
   139→          {category.issues.map((issue, i) => (
   140→            <li key={i} className="flex items-start gap-2 text-xs text-muted-foreground font-[family-name:var(--font-inter)]">
   141→              <AlertTriangle className="size-3.5 text-amber-500 mt-0.5 shrink-0" />
   142→              <span>{issue}</span>
   143→            </li>
   144→          ))}
   145→        </ul>
   146→      )}
   147→    </div>
   148→  );
   149→}
