    82→    82→    82→
    83→    83→    83→const TABLET_BREAKPOINT = 1024;
    84→    84→    84→
    85→    85→    85→function useIsTablet() {
    86→    86→    86→  const [isTablet, setIsTablet] = useState(false);
    87→    87→    87→  useEffect(() => {
    88→    88→    88→    const update = () => {
    89→    89→    89→      const w = window.innerWidth;
    90→    90→    90→      setIsTablet(w >= 768 && w < TABLET_BREAKPOINT);
    91→    91→    91→    };
    92→    92→    92→    const mql = window.matchMedia(`(min-width: 768px) and (max-width: ${TABLET_BREAKPOINT - 1}px)`);
    93→    93→    93→    mql.addEventListener('change', () => update());
    94→    94→    94→    update();
    95→    95→    95→    return () => mql.removeEventListener('change', () => update());
    96→    96→    96→  }, []);
    97→    97→    97→  return isTablet;
    98→    98→    98→}
