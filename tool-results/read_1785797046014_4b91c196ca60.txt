    99→// ── Helpers ──────────────────────────────────────────────────────────────────

   101→const PLAN_ORDER = ['free', 'pro', 'enterprise'] as const;
   102→type PlanName = (typeof PLAN_ORDER)[number];
   103→
   104→function getPlanTier(planName: string): number {
   105→  const idx = PLAN_ORDER.indexOf(planName.toLowerCase() as PlanName);
   106→  return idx >= 0 ? idx : 0;
   107→}
   108→
   109→function formatPrice(cents: number, currency: string = 'USD'): string {
   110→  const amount = cents / 100;
   111→  return new Intl.NumberFormat('en-US', {
   112→    style: 'currency',
   113→    currency,
   114→    minimumFractionDigits: 2,
   115→    maximumFractionDigits: 2,
   116→  }).format(amount);
   117→}
   118→
   119→function formatDate(dateStr: string | null): string {
   120→  if (!dateStr) return 'N/A';
   121→  return new Date(dateStr).toLocaleDateString('en-US', {
   122→    year: 'numeric',
   123→    month: 'long',
   124→    day: 'numeric',
   125→  });
   126→}
   127→
   128→function formatShortDate(dateStr: string | null): string {
   129→  if (!dateStr) return '—';
   130→  return new Date(dateStr).toLocaleDateString('en-US', {
   131→    year: 'numeric',
   132→    month: 'short',
   133→    day: 'numeric',
   134→  });
   135→}
   136→
   137→function getFeatureLabel(key: string): string {
   138→  return FEATURE_DISPLAY_INFO[key]?.label ?? key.replace(/_/g, ' ');
   139→}
   140→
   141→function getFeatureLimitText(feature: PlanFeature): string {
   142→  if (!feature.enabled) return '';
   143→  if (feature.limit === null || feature.limit === -1 || feature.limit === 0) return 'Unlimited';
   144→  return String(feature.limit);
   145→}

   266→        method: 'POST',
   267→        credentials: 'include',

   269→        body: JSON.stringify({ planId, billingCycle: isYearly ? 'yearly' : 'monthly' }),
   270→      });

   273→      if (data.success) {
   274→        if (data.payment_link) {
   275→          window.location.href = data.payment_link;
   276→        } else {
   277→          toast.success(data.message || `Upgraded to ${planDisplayName}!`);
   278→          await fetchAllData();
   279→        }
   280→      } else {
   281→        toast.error(data.error || 'Upgrade failed');

   284→      toast.error('Something went wrong. Please try again.');

   286→      setUpgrading(null);
   287→    }

   295→        credentials: 'include',

   301→      if (data.success) {

   306→        toast.error(data.error || 'Failed to cancel');
   307→      }
   308→    } catch {
   309→      toast.error('Something went wrong. Please try again.');

   312→    }
   313→  };

   321→          <Skeleton className="h-8 w-48" />

   347→          Billing & Plans

   349→        <p className="text-muted-foreground mt-1 text-sm font-body">
   350→          Manage your subscription, view usage, and billing history.
   351→        </p>
   352→      </FadeInUp>
   353→
   354→      {/* ── 1. Current Plan Section ────────────────────────────────────────── */}

   356→        <Card className="overflow-hidden">
   357→          <CardContent className="p-6 md:p-8">
   358→            <div className="flex flex-col md:flex-row md:items-center md:justify-between gap-4">

   363→                  </span>
   364→                </div>

   366→                  <div className="flex items-center gap-2.5">
   367→                    <h2 className="text-lg font-semibold font-heading text-foreground">
   368→                      {subscription?.plan.display_name || 'Free Plan'}

   370→                    <Badge

   373→                        !isCancelled && !isFree

   390→                  </p>
   391→                </div>
   392→              </div>
   393→              <div className="flex items-center gap-3">
   394→                {isFree ? (

   396→                    className="bg-brand text-white hover:bg-brand/90 font-body rounded-lg"
   397→                    onClick={() => {

   441→            {/* Free plan usage bars */}

   556→            <span
