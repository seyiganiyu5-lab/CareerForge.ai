   165→   165→
   166→   166→export function AiAssistPanel({
   167→   167→  open,
   168→   168→  onOpenChange,
   169→   169→  resumeId,
   170→   170→  section,
   171→   171→  content,
   172→   172→  itemId,
   173→   173→  field,
   174→   174→  resumeData,
   175→   175→  onApply,
   176→   176→  onInsertBelow,
   177→   177→}: AiAssistPanelProps) {
   178→   178→  // ---- i18n ----
   179→   179→  const t = useTranslations('ai.assistant');
   180→   180→
   181→   181→  // ---- State ----
   182→   182→  const [results, setResults] = useState<ResultItem[]>([]);
   183→   183→  const [customPrompt, setCustomPrompt] = useState('');
   184→   184→  const [isSending, setIsSending] = useState(false);
   185→   185→  const [loadingMessage, setLoadingMessage] = useState(LOADING_MESSAGES[0]);
   186→   186→  const scrollRef = useRef<HTMLDivElement>(null);
   187→   187→  const requestInProgressRef = useRef(false);
   188→   188→
   189→   189→  const actions = SECTION_ACTIONS[section] ?? [
   190→   190→    { action: 'improve', icon: Lightbulb, label: 'Improve' },
   191→   191→    { action: 'rewrite', icon: RotateCcw, label: 'Rewrite' },
   192→   192→    { action: 'grammar', icon: ShieldCheck, label: 'Grammar Check' },
   193→   193→  ];
   194→   194→
   195→   195→  const sectionDisplayName = SECTION_DISPLAY_NAMES[section] ?? section;
   196→   196→
   197→   197→  // ---- Loading message cycling ----
   198→   198→  useEffect(() => {
   199→   199→    if (!isSending) return;
   200→   200→    let index = 0;
   201→   201→    const interval = setInterval(() => {
   202→   202→      index = (index + 1) % LOADING_MESSAGES.length;
   203→   203→      setLoadingMessage(LOADING_MESSAGES[index]);
   204→   204→    }, 2000);
   205→   205→    return () => clearInterval(interval);
   206→   206→  }, [isSending]);
   207→   207→
   208→   208→  // ---- Clear state when panel closes ----
   209→   209→  const handleOpenChange = useCallback(
   210→   210→    (nextOpen: boolean) => {
   211→   211→      if (!nextOpen) {
   212→   212→        setResults([]);
   213→   213→        setCustomPrompt('');
   214→   214→        setIsSending(false);
   215→   215→        setLoadingMessage(LOADING_MESSAGES[0]);
   216→   216→        requestInProgressRef.current = false;
   217→   217→      }
   218→   218→      onOpenChange(nextOpen);
   219→   219→    },
   220→   220→    [onOpenChange],
   221→   221→  );
   222→   222→
   223→   223→  // ---- Send AI request ----
   224→   224→  const sendAiRequest = useCallback(
   225→   225→    async (action: string, customPromptText?: string) => {
   226→   226→      if (requestInProgressRef.current) return;
   227→   227→      requestInProgressRef.current = true;
   228→   228→      setIsSending(true);
   229→   229→
   230→   230→      const resultId = Date.now().toString(36);
   231→   231→      setResults((prev) => [
   232→   232→        ...prev,
   233→   233→        { id: resultId, text: '', action, loading: true, copied: false },
   234→   234→      ]);
   235→   235→
   236→   236→      // Scroll to bottom after the new result is rendered
   237→   237→      setTimeout(() => {
   238→   238→        scrollRef.current?.scrollTo({
   239→   239→          top: scrollRef.current.scrollHeight,
   240→   240→          behavior: 'smooth',
   241→   241→        });
   242→   242→      }, 100);
   243→   243→
   244→   244→      try {
   245→   245→        const res = await fetch(`/api/resumes/${resumeId}/ai-assist`, {
   246→   246→          method: 'POST',
   247→   247→          headers: { 'Content-Type': 'application/json' },
   248→   248→          credentials: 'include',
   249→   249→          body: JSON.stringify({
   250→   250→            section,
   251→   251→            action,
   252→   252→            content,
   253→   253→            itemId,
   254→   254→            field,
   255→   255→            customPrompt: customPromptText,
   256→   256→            resumeContext: resumeData,
   257→   257→          }),
   258→   258→        });
   259→   259→
   260→   260→        const data = await res.json();
   261→   261→
   262→   262→        if (!res.ok) {
   263→   263→          if (data.error === 'limit_reached' || data.error === 'upgrade_required') {
   264→   264→            setResults((prev) => prev.filter((r) => r.id !== resultId));
   265→   265→            onOpenChange(false);
   266→   266→            window.dispatchEvent(
   267→   267→              new CustomEvent('ai-limit-reached', { detail: data }),
   268→   268→            );
   269→   269→            return;
   270→   270→          }
   271→   271→          throw new Error(data.error || 'Failed to generate suggestion');
   272→   272→        }
   273→   273→
   274→   274→        setResults((prev) =>
   275→   275→          prev.map((r) =>
   276→   276→            r.id === resultId
   277→   277→              ? { ...r, text: data.result, loading: false }
   278→   278→              : r,
   279→   279→          ),
   280→   280→        );
   281→   281→      } catch (err) {
   282→   282→        setResults((prev) =>
   283→   283→          prev.map((r) =>
   284→   284→            r.id === resultId
   285→   285→              ? {
   286→   286→                  ...r,
   287→   287→                  loading: false,
   288→   288→                  error:
   289→   289→                    err instanceof Error
   290→   290→                      ? err.message
   291→   291→                      : 'An error occurred',
   292→   292→                }
   293→   293→              : r,
   294→   294→          ),
   295→   295→        );
   296→   296→      } finally {
   297→   297→        setIsSending(false);
   298→   298→        requestInProgressRef.current = false;
   299→   299→      }
   300→   300→    },
   301→   301→    [resumeId, section, content, itemId, field, resumeData, onOpenChange],
   302→   302→  );
   303→   303→
   304→   304→  // ---- Copy to clipboard ----
   305→   305→  const copyToClipboard = useCallback(
   306→   306→    async (text: string, resultId: string) => {
   307→   307→      await navigator.clipboard.writeText(text);
   308→   308→      setResults((prev) =>
   309→   309→        prev.map((r) => (r.id === resultId ? { ...r, copied: true } : r)),
   310→   310→      );
   311→   311→      setTimeout(() => {
   312→   312→        setResults((prev) =>
   313→   313→          prev.map((r) =>
   314→   314→            r.id === resultId ? { ...r, copied: false } : r,
   315→   315→          ),
   316→   316→        );
   317→   317→      }, 2000);
   318→   318→    },
   319→   319→    [],
   320→   320→  );
   321→   321→
   322→   322→  // ---- Apply result ----
   323→   323→  const handleApply = useCallback(
   324→   324→    (text: string) => {
   325→   325→      onApply(text);
   326→   326→      onOpenChange(false);
   327→   327→    },
   328→   328→    [onApply, onOpenChange],
   329→   329→  );
   330→   330→
   331→   331→  // ---- Handle custom prompt submit ----
   332→   332→  const handleCustomSubmit = useCallback(() => {
   333→   333→    const trimmed = customPrompt.trim();
   334→   334→    if (!trimmed || requestInProgressRef.current) return;
   335→   335→    sendAiRequest('custom', trimmed);
   336→   336→    setCustomPrompt('');
   337→   337→  }, [customPrompt, sendAiRequest]);
   338→   338→
   339→   339→  // ---- Handle action button click ----
   340→   340→  const handleActionClick = useCallback(
   341→   341→    (action: string) => {
   342→   342→      if (requestInProgressRef.current) return;
   343→   343→      sendAiRequest(action);
   344→   344→    },
   345→   345→    [sendAiRequest],
   346→   346→  );
   347→   347→
   348→   348→  return (
   349→   349→    <Sheet open={open} onOpenChange={handleOpenChange}>
   350→   350→      <TooltipProvider delayDuration={300}>
   351→   351→        <SheetContent
   352→   352→          side="right"
   353→   353→          className="sm:max-w-[480px] p-0 gap-0 [&>button.absolute]:hidden"
   354→   354→        >
   355→   355→          {/* =========================================================== */}
   356→   356→          {/* Header                                                      */}
   357→   357→          {/* =========================================================== */}
   358→   358→          <div className="relative flex items-center justify-between border-b border-border px-5 py-4">
   359→   359→            {/* Gradient accent line */}
   360→   360→            <div className="absolute inset-x-0 top-0 h-[3px] bg-gradient-to-r from-brand via-violet to-emerald" />
   361→   361→
   362→   362→            <div className="flex items-center gap-3">
   363→   363→              <div className="flex size-9 items-center justify-center rounded-lg bg-brand/10">
   364→   364→                <Sparkles className="size-5 text-brand" />
   365→   365→              </div>
   366→   366→              <div>
   367→   367→                <SheetTitle className="font-[family-name:var(--font-poppins)] text-base font-semibold text-foreground">
   368→   368→                  AI Resume Assistant
   369→   369→                </SheetTitle>
   370→   370→                <p className="mt-0.5 text-xs text-muted-foreground font-[family-name:var(--font-inter)]">
   371→   371→                  {t('subtitle') || 'Powered by AI'}
   372→   372→                </p>
   373→   373→              </div>
   374→   374→            </div>
   375→   375→
   376→   376→            <Button
   377→   377→              variant="ghost"
   378→   378→              size="icon"
   379→   379→              className="size-8 rounded-md"
   380→   380→              onClick={() => handleOpenChange(false)}
   381→   381→            >
   382→   382→              <X className="size-4" />
   383→   383→              <span className="sr-only">Close</span>
   384→   384→            </Button>
   385→   385→          </div>
   386→   386→
   387→   387→          {/* =========================================================== */}
   388→   388→          {/* Scrollable body                                             */}
   389→   389→          {/* =========================================================== */}
   390→   390→          <div
   391→   391→            ref={scrollRef}
   392→   392→            className="flex-1 overflow-y-auto overscroll-contain"
   393→   393→          >
   394→   394→            {/* --- Section badge & context --- */}
   395→   395→            <div className="px-5 pt-4 pb-2">
   396→   396→              <div className="flex items-center gap-2">
   397→   397→                <Badge
   398→   398→                  variant="secondary"
   399→   399→                  className="bg-brand/10 text-brand border-brand/20 font-[family-name:var(--font-inter)] font-medium text-xs"
   400→   400→                >
   401→   401→                  {sectionDisplayName}
   402→   402→                </Badge>
   403→   403→                {content && (
   404→   404→                  <Badge
   405→   405→                    variant="outline"
   406→   406→                    className="text-muted-foreground font-[family-name:var(--font-inter)] text-[11px]"
   407→   407→                  >
   408→   408→                    {content.length > 50
   409→   409→                      ? `${content.length} chars`
   410→   410→                      : `"${content.slice(0, 50)}${content.length > 50 ? '...' : ''}"`}
   411→   411→                  </Badge>
   412→   412→                )}
   413→   413→              </div>
   414→   414→              <p className="mt-3 text-sm text-muted-foreground font-[family-name:var(--font-inter)] leading-relaxed">
   415→   415→                {t('description') ||
   416→   416→                  'Select a quick action below or type a custom request to enhance this section.'}
   417→   417→              </p>
   418→   418→            </div>
   419→   419→
   420→   420→            <Separator className="mx-5 w-auto" />
   421→   421→
   422→   422→            {/* --- Quick actions grid --- */}
   423→   423→            <div className="px-5 pt-4 pb-2">
   424→   424→              <p className="mb-3 text-xs font-medium uppercase tracking-wider text-muted-foreground font-[family-name:var(--font-inter)]">
   425→   425→                {t('quickActions') || 'Quick Actions'}
   426→   426→              </p>
   427→   427→              <div className="grid grid-cols-2 gap-2">
   428→   428→                <AnimatePresence mode="popLayout">
   429→   429→                  {actions.map((item) => {
   430→   430→                    const Icon = item.icon;
   431→   431→                    return (
   432→   432→                      <motion.button
   433→   433→                        key={item.action}
   434→   434→                        type="button"
   435→   435→                        whileHover={{ scale: 1.03 }}
   436→   436→                        whileTap={{ scale: 0.97 }}
   437→   437→                        onClick={() => handleActionClick(item.action)}
   438→   438→                        disabled={isSending}
   439→   439→                        className="group flex items-center gap-2.5 rounded-lg border border-border bg-card px-3 py-2.5 text-left text-sm font-medium text-foreground transition-colors hover:border-brand/40 hover:bg-brand/5 disabled:pointer-events-none disabled:opacity-50 font-[family-name:var(--font-inter)]"
   440→   440→                      >
   441→   441→                        <Icon className="size-4 shrink-0 text-muted-foreground transition-colors group-hover:text-brand" />
   442→   442→                        <span className="truncate">{item.label}</span>
   443→   443→                      </motion.button>
   444→   444→                    );
   445→   445→                  })}
   446→   446→                </AnimatePresence>
   447→   447→              </div>
   448→   448→            </div>
   449→   449→
   450→   450→            <Separator className="mx-5 w-auto" />
   451→   451→
   452→   452→            {/* --- Custom prompt --- */}
   453→   453→            <div className="px-5 pt-4 pb-2">
   454→   454→              <p className="mb-3 text-xs font-medium uppercase tracking-wider text-muted-foreground font-[family-name:var(--font-inter)]">
   455→   455→                {t('customPrompt') || 'Or ask AI anything'}
   456→   456→              </p>
   457→   457→              <div className="relative">
   458→   458→                <Textarea
   459→   459→                  value={customPrompt}
   460→   460→                  onChange={(e) => setCustomPrompt(e.target.value)}
   461→   461→                  placeholder={
   462→   462→                    t('placeholder') ||
   463→   463→                    'Type your request... e.g., "Make it more concise and action-oriented"'
   464→   464→                  }
   465→   465→                  rows={3}
   466→   466→                  className="resize-none rounded-lg border-border bg-card pr-12 text-sm font-[family-name:var(--font-inter)] placeholder:text-muted-foreground/60 focus-visible:ring-brand/30"
   467→   467→                  onKeyDown={(e) => {
   468→   468→                    if (e.key === 'Enter' && (e.metaKey || e.ctrlKey)) {
   469→   469→                      e.preventDefault();
   470→   470→                      handleCustomSubmit();
   471→   471→                    }
   472→   472→                  }}
   473→   473→                />
   474→   474→                <Tooltip>
   475→   475→                  <TooltipTrigger asChild>
   476→   476→                    <Button
   477→   477→                      size="icon"
   478→   478→                      onClick={handleCustomSubmit}
   479→   479→                      disabled={!customPrompt.trim() || isSending}
   480→   480→                      className="absolute bottom-2 right-2 size-8 rounded-md bg-brand text-primary-foreground shadow-sm hover:bg-brand/90 disabled:opacity-40"
   481→   481→                    >
   482→   482→                      {isSending ? (
   483→   483→                        <Loader2 className="size-4 animate-spin" />
   484→   484→                      ) : (
   485→   485→                        <Send className="size-4" />
   486→   486→                      )}
   487→   487→                      <span className="sr-only">Send</span>
   488→   488→                    </Button>
   489→   489→                  </TooltipTrigger>
