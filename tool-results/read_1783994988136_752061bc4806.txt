    87→    87→const TABLET_BREAKPOINT = 1024;
    88→    88→
    89→    89→function useIsTablet() {
    90→    90→  const [isTablet, setIsTablet] = useState(false);
    91→    91→  useEffect(() => {
    92→    92→    const update = () => {
    93→    93→      const w = window.innerWidth;
    94→    94→      setIsTablet(w >= 768 && w < TABLET_BREAKPOINT);
    95→    95→    };
    96→    96→    const mql = window.matchMedia(`(min-width: 768px) and (max-width: ${TABLET_BREAKPOINT - 1}px)`);
    97→    97→    mql.addEventListener('change', () => update());
    98→    98→    update();
    99→    99→    return () => mql.removeEventListener('change', () => update());
   100→   100→  }, []);
   101→   101→  return isTablet;
   102→   102→}
