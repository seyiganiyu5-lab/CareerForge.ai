   220→   220→function scoreWorkExperience(text: string, data: ResumeData | null): CategoryResult {
   221→   221→  const issues: string[] = [];
   222→   222→  let score = 0;
   223→   223→  const maxScore = 20;
   224→   224→
   225→   225→  const entries = data?.workExperience ?? [];
   226→   226→
   227→   227→  if (entries.length === 0) {
   228→   228→    issues.push('Work experience section is missing. This is one of the most important sections for ATS.');
   229→   229→    return { score: 0, maxScore, label: 'Work Experience', issues };
   230→   230→  }
   231→   231→
   232→   232→  // Up to 5 points for having entries
   233→   233→  score += Math.min(5, entries.length * 3);
   234→   234→
   235→   235→  let hasGoodEntry = false;
   236→   236→  let goodEntryCount = 0;
   237→   237→
   238→   238→  for (const entry of entries) {
   239→   239→    let entryScore = 0;
   240→   240→
   241→   241→    if (entry.company?.trim()) entryScore += 1;
   242→   242→    else issues.push(`Work entry is missing a company name.`);
   243→   243→
   244→   244→    if (entry.position?.trim()) entryScore += 1;
   245→   245→    else issues.push(`Work entry is missing a job title/position.`);
   246→   246→
   247→   247→    if (entry.startDate?.trim()) entryScore += 1;
   248→   248→    else issues.push(`Work entry is missing a start date.`);
   249→   249→
   250→   250→    // Description with bullet-like content
   251→   251→    const desc = entry.description?.trim() ?? '';
   252→   252→    if (desc.length > 0) {
   253→   253→      entryScore += 1;
   254→   254→      // Check for quantifiable results or bullet-like items
   255→   255→      const bulletIndicators = /^[-•●▪▶◆]\s|^•\s|\n[-•●▪▶◆]\s|\n•\s/m;
   256→   256→      const hasBullets = bulletIndicators.test(desc);
   257→   257→      const hasNumbers = /\d+%|\$[\d,]+|\d+\+?\s*(years?|users?|clients?|projects?|team|people|members?)/i.test(desc);
   258→   258→      if (hasBullets) entryScore += 1;
   259→   259→      if (hasNumbers) entryScore += 1;
   260→   260→    } else {
   261→   261→      issues.push(`Work entry is missing a description. Add bullet points with achievements.`);
   262→   262→    }
   263→   263→
   264→   264→    if (entryScore >= 4) {
   265→   265→      hasGoodEntry = true;
   266→   266→      goodEntryCount++;
   267→   267→    }
   268→   268→  }
   269→   269→
   270→   270→  // Quality points
   271→   271→  if (hasGoodEntry) score += 5;
   272→   272→  else issues.push('No work experience entry has all required fields (company, position, dates, description).');
   273→   273→
   274→   274→  if (goodEntryCount >= 2) score += 3;
   275→   275→  else if (entries.length >= 2) issues.push('Ensure all work experience entries have complete details.');
   276→   276→
   277→   277→  // Points for descriptions with impact
   278→   278→  const allDescs = entries.map(e => e.description ?? '').join(' ');
   279→   279→  const actionVerbs = /managed|led|developed|achieved|improved|built|delivered|drove|increased|reduced|implemented|created|designed|launched|optimized|coordinated|established|generated|streamlined|spearheaded/i;
   280→   280→  if (actionVerbs.test(allDescs)) score += 2;
   281→   281→  else issues.push('Use action verbs (e.g., "led", "developed", "achieved") in your work descriptions.');
   282→   282→
   283→   283→  if (score > maxScore) score = maxScore;
   284→   284→
   285→   285→  return { score, maxScore, label: 'Work Experience', issues };
   286→   286→}

   362→   362→function scoreFormatting(text: string, data: ResumeData | null): CategoryResult {
   363→   363→  const issues: string[] = [];
   364→   364→  let score = 0;
   365→   365→  const maxScore = 15;
   366→   366→
   367→   367→  // Check for section headers
   368→   368→  const expectedHeaders = [
   369→   369→    'experience', 'education', 'skill',
   370→   370→  ];
   371→   371→  const textLower = text.toLowerCase();
   372→   372→  const headerPattern = new RegExp(expectedHeaders.map(h => `\\b${h}\\b`).join('|'), 'i');
   373→   373→  const headersFound = (textLower.match(headerPattern) || []).length;
   374→   374→  if (headersFound >= 3) score += 4;
   375→   375→  else if (headersFound >= 2) { score += 3; issues.push('Consider adding more standard section headers (Experience, Education, Skills).'); }
   376→   376→  else issues.push('Standard section headers (Experience, Education, Skills) are missing or hard to detect.');
   377→   377→
   378→   378→  // Check for consistent date formats within work experience
   379→   379→  const dates = data?.workExperience
   380→   380→    ?.flatMap(e => [e.startDate, e.endDate].filter(Boolean))
   381→   381→    ?? [];
   382→   382→
   383→   383→  if (dates.length >= 2) {
   384→   384→    // If using structured data, dates are likely consistent
   385→   385→    score += 3;
   386→   386→  } else if (dates.length > 0) {
   387→   387→    score += 2;
   388→   388→  } else {
   389→   389→    // Check raw text for dates
   390→   390→    const datePatterns = text.match(/(?:Jan|Feb|Mar|Apr|May|Jun|Jul|Aug|Sep|Oct|Nov|Dec)[a-z]*\.? \d{4}|\d{1,2}\/\d{1,2}\/\d{2,4}|\d{4}\s*[-–—]\s*\d{4}|(?:January|February|March|April|May|June|July|August|September|October|November|December)\s+\d{4}/gi) || [];
   391→   391→    if (datePatterns.length >= 2) score += 2;
   392→   392→    else issues.push('Add employment dates to your work experience entries.');
   393→   393→  }
   394→   394→
   395→   395→  // Check for bullet point usage in descriptions
   396→   396→  const allDescriptions = data?.workExperience?.map(e => e.description ?? '').join('\n') ?? '';
   397→   397→  const bulletCount = (allDescriptions.match(/^[-•●▪▶◆]\s/gm) || []).length;
   398→   398→  if (bulletCount >= 3) score += 3;
   399→   399→  else if (bulletCount >= 1) { score += 2; issues.push('Use more bullet points to describe your achievements and responsibilities.'); }
   400→   400→  else if (allDescriptions.length > 0) issues.push('Use bullet points (•, -, etc.) to format your experience descriptions.');
   401→   401→
   402→   402→  // Check for special characters that can confuse ATS
   403→   403→  const atsUnfriendlyChars = /[│║═╔╗╚╝┌┐└┘├┤┬┴┼]/;
   404→   404→  if (!atsUnfriendlyChars.test(text)) score += 2;
   405→   405→  else issues.push('Remove box-drawing or special characters that ATS systems cannot parse.');
   406→   406→
   407→   407→  // Check that text is not all caps (which can look unprofessional)
   408→   408→  const words = text.split(/\s+/).filter(w => w.length > 1);
   409→   409→  if (words.length > 0) {
   410→   410→    const allCapsWords = words.filter(w => w === w.toUpperCase() && /[A-Z]/.test(w));
   411→   411→    const allCapsRatio = allCapsWords.length / words.length;
   412→   412→    if (allCapsRatio < 0.1) score += 3;
   413→   413→    else if (allCapsRatio < 0.3) { score += 2; issues.push('Avoid using excessive ALL CAPS text. Reserve uppercase for section headers only.'); }
   414→   414→    else { score += 1; issues.push('Too much uppercase text detected. Use standard capitalization for better readability.'); }
   415→   415→  } else {
   416→   416→    score += 3;
   417→   417→  }
   418→   418→
   419→   419→  if (score > maxScore) score = maxScore;
   420→   420→
   421→   421→  return { score, maxScore, label: 'Formatting Quality', issues };
   422→   422→}

   424→   424→function scoreLength(text: string): CategoryResult {

   438→   438→  } else if (wordCount < 300) {

   440→   440→    score = 5;
   441→   441→  } else if (wordCount <= 1500) {

   443→   443→    score = 10;
   444→   444→  } else if (wordCount <= 2000) {
   445→   445→    issues.push('Resume is slightly long. Consider trimming to keep it within 2 pages.');
   446→   446→    score = 7;
   447→   447→  } else {
   448→   448→    issues.push('Resume is too long (over 2000 words). Most recruiters prefer 1-2 page resumes. Trim less relevant content.');
   449→   449→    score = 4;
   450→   450→  }

   452→   452→  return { score, maxScore, label: 'Resume Length', issues };

   455→   455→function extractKeywords(jobDescription: string): string[] {
   456→   456→  // Common stop words to exclude from keywords
   457→   457→  const stopWords = new Set([
   458→   458→    'the', 'a', 'an', 'and', 'or', 'but', 'in', 'on', 'at', 'to', 'for',
   459→   459→    'of', 'with', 'by', 'from', 'is', 'are', 'was', 'were', 'be', 'been',

   465→   465→    'also', 'then', 'there', 'here', 'each', 'every', 'all', 'both',
   466→   466→    'few', 'more', 'most', 'other', 'some', 'such', 'only', 'own',
   467→   467→    'same', 'what', 'which', 'who', 'whom', 'how', 'when', 'where',
   468→   468→    'why', 'while', 'up', 'out', 'into', 'through', 'during', 'before',
   469→   469→    'after', 'above', 'below', 'between', 'under', 'again', 'further',
   470→   470→    'once', 'work', 'working', 'experience', 'required', 'requirements',
   471→   471→    'preferred', 'qualifications', 'skills', 'ability', 'able', 'strong',
   472→   472→    'excellent', 'good', 'great', 'looking', 'join', 'team', 'role',
   473→   473→    'position', 'job', 'company', 'opportunity', 'responsibilities',
   474→   474→    'include', 'including', 'well', 'new', 'plus', 'etc', 'e.g.',
   475→   475→    'i.e.', 'per', 'via', 'within', 'across', 'among', 'along',
   476→   476→    'minimum', 'maximum', 'least', 'years', 'year', 'like',
   477→   477→  ]);
   478→   478→
   479→   479→  // Extract multi-word technical terms first (2-3 words)
   480→   480→  const techPhrases: string[] = [];
   481→   481→  const phrasePatterns = [
   482→   482→    /machine\s+learning/gi,
   483→   483→    /deep\s+learning/gi,
   484→   484→    /data\s+science/gi,
   485→   485→    /data\s+analysis/gi,
   486→   486→    /project\s+management/gi,
   487→   487→    /product\s+management/gi,
   488→   488→    /full\s+stack/gi,
   489→   489→    /front\s*[- ]?end/gi,
   490→   490→    /back\s*[- ]?end/gi,
   491→   491→    /ci\s*\/\s*cd/gi,
   492→   492→    /cloud\s+computing/gi,
   493→   493→    /software\s+development/gi,
   494→   494→    /agile\s+methodology/gi,
   495→   495→    /rest\s*[- ]?api/gi,
   496→   496→    /graphql/gi,
   497→   497→    /user\s+experience/gi,
   498→   498→    /ui\s*\/\s*ux/gi,
   499→   499→    /natural\s+language\s+processing/gi,
   500→   500→    /computer\s+vision/gi,
   501→   501→    /big\s+data/gi,
   502→   502→    /supply\s+chain/gi,
   503→   503→    /quality\s+assurance/gi,
   504→   504→    /continuous\s+integration/gi,
   505→   505→    /continuous\s+delivery/gi,
   506→   506→    /cross[\s-]functional/gi,
   507→   507→    /stakeholder\s+management/gi,
   508→   508→    /business\s+development/gi,
   509→   509→    /customer\s+experience/gi,
   510→   510→    /a\/b\s+testing/gi,
   511→   511→    /unit\s+testing/gi,
   512→   512→  ];
   513→   513→
   514→   514→  for (const pattern of phrasePatterns) {
   515→   515→    const matches = jobDescription.match(pattern);
   516→   516→    if (matches) {
   517→   517→      techPhrases.push(...matches.map(m => m.trim().toLowerCase()));
   518→   518→    }
   519→   519→  }
   520→   520→
   521→   521→  // Extract single words (alphabetic, 2+ chars)
   522→   522→  const singleWords = jobDescription
   523→   523→    .replace(/[^a-zA-Z\s\-\/\.]/g, ' ')
   524→   524→    .split(/\s+/)
   525→   525→    .map(w => w.toLowerCase().replace(/^[-\/\.]+|[-\/\.]+$/g, ''))
   526→   526→    .filter(w => w.length >= 2 && !stopWords.has(w) && !/^\d+$/.test(w));
   527→   527→
   528→   528→  // Count frequency
   529→   529→  const wordFreq = new Map<string, number>();
   530→   530→  for (const w of singleWords) {
   531→   531→    wordFreq.set(w, (wordFreq.get(w) ?? 0) + 1);
   532→   532→  }
   533→   533→
   534→   534→  // Get keywords that appear at least once (or are tech phrases)
   535→   535→  const keywords = new Set<string>();
   536→   536→  for (const [word, count] of wordFreq) {
   537→   537→    if (count >= 1 && word.length >= 3) {
   538→   538→      keywords.add(word);
   539→   539→    }
   540→   540→  }
   541→   541→  for (const phrase of techPhrases) {
   542→   542→    keywords.add(phrase);
   543→   543→  }
   544→   544→
   545→   545→  return Array.from(keywords);
   546→   546→}

   548→   548→function scoreKeywords(
   549→   549→  resumeText: string,
   550→   550→  jobDescription: string,
   551→   551→): { category: CategoryResult; matchResult: KeywordMatchResult } {
   552→   552→  const issues: string[] = [];
   553→   553→  let score = 0;
   554→   554→  const maxScore = 10;
   555→   555→
   556→   556→  const keywords = extractKeywords(jobDescription);
   557→   557→  const resumeLower = resumeText.toLowerCase();
   558→   558→
   559→   559→  const matched: string[] = [];
   560→   560→  const missing: string[] = [];
   561→   561→
   562→   562→  for (const kw of keywords) {
   563→   563→    if (resumeLower.includes(kw.toLowerCase())) {
   564→   564→      matched.push(kw);
   565→   565→    } else {
   566→   566→      missing.push(kw);
   567→   567→    }
   568→   568→  }
   569→   569→
   570→   570→  // Cap missing display to 10 most relevant
   571→   571→  const displayMissing = missing.slice(0, 10);
   572→   572→
   573→   573→  const matchRate = keywords.length > 0 ? matched.length / keywords.length : 1;
   574→   574→
   575→   575→  if (keywords.length === 0) {
   576→   576→    score = 10;
   577→   577→  } else if (matchRate >= 0.7) {
   578→   578→    score = 10;
   579→   579→  } else if (matchRate >= 0.5) {
   580→   580→    score = 7;
   581→   581→    if (displayMissing.length > 0) {
   582→   582→      issues.push(`${matched.length} of ${keywords.length} job keywords found. Consider adding: ${displayMissing.join(', ')}`);
   583→   583→    }
   584→   584→  } else if (matchRate >= 0.3) {
   585→   585→    score = 5;
   586→   586→    if (displayMissing.length > 0) {
   587→   587→      issues.push(`${matched.length} of ${keywords.length} job keywords found. Consider adding: ${displayMissing.join(', ')}`);
   588→   588→    }
   589→   589→  } else if (matchRate >= 0.15) {
   590→   590→    score = 3;
   591→   591→    if (displayMissing.length > 0) {
   592→   592→      issues.push(`Only ${matched.length} of ${keywords.length} job keywords found. Consider adding: ${displayMissing.join(', ')}`);
   593→   593→    }
   594→   594→  } else {
   595→   595→    score = 1;
   596→   596→    if (displayMissing.length > 0) {
   597→   597→      issues.push(`Very few keywords matched (${matched.length}/${keywords.length}). Strongly consider incorporating: ${displayMissing.join(', ')}`);
   598→   598→    }
   599→   599→  }
   600→   600→
   601→   601→  return {
   602→   602→    category: { score, maxScore, label: 'Keyword Match', issues },
   603→   603→    matchResult: {
   604→   604→      matched,
   605→   605→      missing: displayMissing,
   606→   606→      matchRate: Math.round(matchRate * 100) / 100,
   607→   607→    },
   608→   608→  };
   609→   609→}

   615→   615→async function runAiAnalysis(resumeText: string): Promise<AiAnalysis> {
   616→   616→  const zai = await ZAI.create();
   617→   617→
   618→   618→  const systemPrompt = `You are an expert ATS (Applicant Tracking System) resume analyst. Analyze the given resume text and provide detailed, actionable feedback.
   619→   619→
   620→   620→IMPORTANT RULES:
   621→   621→- NEVER mention or suggest changes to date formats. All common date formats (Jan 2023, 01/2023, 2023-01, January 2023, etc.) are perfectly valid.
   622→   622→- Focus on content quality, impact, structure, and ATS optimization.
   623→   623→- Be specific and actionable in your suggestions.
   624→   624→
   625→   625→Respond with a JSON object (no markdown, no code fences, pure JSON) with this exact structure:
   626→   626→{
   627→   627→  "grammarIssues": [{"message": "Description of the issue", "suggestion": "How to fix it"}],
   628→   628→  "suggestions": [{"message": "What to improve", "priority": "high|medium|low"}],
   629→   629→  "strengths": ["List of 3-5 strengths"],
   630→   630→  "weaknesses": ["List of 3-5 weakness areas"]
   631→   631→}
   632→   632→
   633→   633→Guidelines:
   634→   634→- grammarIssues: Find 0-5 real grammar, spelling, or phrasing issues
   635→   635→- suggestions: Provide 3-6 actionable improvement suggestions with priority levels
   636→   636→- strengths: Highlight 3-5 specific things done well
   637→   637→- weaknesses: Identify 3-5 specific areas needing improvement`;
   638→   638→
   639→   639→  const completion = await zai.chat.completions.create({
   640→   640→    messages: [
   641→   641→      { role: 'system', content: systemPrompt },
   642→   642→      { role: 'user', content: `Analyze this resume for ATS optimization:\n\n${resumeText}` },
   643→   643→    ],
   644→   644→    thinking: { type: 'disabled' },
   645→   645→  });
   646→   646→
   647→   647→  const raw = completion.choices[0]?.message?.content?.trim() ?? '';
   648→   648→
   649→   649→  // Try to parse the JSON response
   650→   650→  try {
   651→   651→    // Strip any markdown code fences if present
   652→   652→    const cleaned = raw.replace(/^```(?:json)?\s*\n?/i, '').replace(/\n?```\s*$/i, '').trim();
   653→   653→    const parsed = JSON.parse(cleaned) as AiAnalysis;
   654→   654→
   655→   655→    // Validate and sanitize
   656→   656→    return {
   657→   657→      grammarIssues: Array.isArray(parsed.grammarIssues)
   658→   658→        ? parsed.grammarIssues.slice(0, 8).filter(i => i.message)
   659→   659→        : [],
   660→   660→      suggestions: Array.isArray(parsed.suggestions)
   661→   661→        ? parsed.suggestions.slice(0, 8).filter(s => s.message && ['high', 'medium', 'low'].includes(s.priority))
   662→   662→        : [],
   663→   663→      strengths: Array.isArray(parsed.strengths) ? parsed.strengths.slice(0, 6) : [],
   664→   664→      weaknesses: Array.isArray(parsed.weaknesses) ? parsed.weaknesses.slice(0, 6) : [],
   665→   665→    };
   666→   666→  } catch {
   667→   667→    // If parsing fails, return empty analysis
   668→   668→    return {
   669→   669→      grammarIssues: [],
   670→   670→      suggestions: [],
   671→   671→      strengths: ['Resume content was received and processed.'],
   672→   672→      weaknesses: ['Unable to generate detailed AI analysis at this time.'],
   673→   673→    };
   674→   674→  }
   675→   675→}
