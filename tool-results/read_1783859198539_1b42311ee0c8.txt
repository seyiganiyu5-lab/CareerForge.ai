   483→function estimateYearsOfExperience(workExperience?: ResumeContext['workExperience']): string {
   484→  if (!workExperience || workExperience.length === 0) return 'several';
   485→  let totalYears = 0;
   486→  for (const exp of workExperience) {
   487→    if (exp.startDate) {
   488→      const start = new Date(exp.startDate);
   489→      const end = exp.current ? new Date() : (exp.endDate ? new Date(exp.endDate) : new Date());
   490→      const diffMs = end.getTime() - start.getTime();
   491→      totalYears += Math.max(0, Math.round(diffMs / (365.25 * 24 * 60 * 60 * 1000)));
   492→    }
   493→  }
   494→  return totalYears > 0 ? String(totalYears) : 'several';
   495→}

   521→// ---------------------------------------------------------------------------
   522→// POST Handler
   523→// ---------------------------------------------------------------------------
   524→
   525→export async function POST(
   526→  request: NextRequest,
   527→  { params }: { params: Promise<{ id: string }> }
   528→) {
   529→  try {
   530→    // 1. Authentication
   531→    const auth = await getAuthUser();
   532→    if (!auth) {
   533→      return NextResponse.json(
   534→        { success: false, error: 'Unauthorized' },
   535→        { status: 401 }
   536→      );
   537→    }
   538→
   539→    // 2. Parse & validate request body
   540→    const body: AiAssistBody = await request.json();
   541→    const { section, action, content, customPrompt, tone, resumeContext, itemId, field } = body;
   542→
   543→    if (!section || !action) {
   544→      return NextResponse.json(
   545→        { success: false, error: 'Missing required fields: section and action' },
   546→        { status: 400 }
   547→      );
   548→    }
   549→
   550→    if (!VALID_SECTIONS.includes(section as Section)) {
   551→      return NextResponse.json(
   552→        { success: false, error: `Invalid section. Must be one of: ${VALID_SECTIONS.join(', ')}` },
   553→        { status: 400 }
   554→      );
   555→    }
   556→
   557→    if (!VALID_ACTIONS.includes(action as Action)) {
   558→      return NextResponse.json(
   559→        { success: false, error: `Invalid action. Must be one of: ${VALID_ACTIONS.join(', ')}` },
   560→        { status: 400 }
   561→      );
   562→    }
   563→
   564→    if (action === 'custom' && !customPrompt) {
   565→      return NextResponse.json(
   566→        { success: false, error: 'customPrompt is required when action is "custom"' },
   567→        { status: 400 }
   568→      );
   569→    }
   570→
   571→    // 3. Resolve resume ID from route params
   572→    const { id } = await params;
   573→
   574→    // 4. Ownership check
   575→    const resume = await db.resume.findUnique({
   576→      where: { id },
   577→    });
   578→
   579→    if (!resume || resume.user_id !== auth.userId) {
   580→      return NextResponse.json(
   581→        { success: false, error: 'Resume not found' },
   582→        { status: 404 }
   583→      );
   584→    }
   585→
   586→    // 5. Feature gating
   587→    const usage = await checkAndIncrementUsage(auth.userId, FEATURE_KEYS.AI_RESUME_BUILDER);
   588→    if (!usage.allowed) {
   589→      return NextResponse.json(
   590→        {
   591→          success: false,
   592→          error: usage.reason || 'limit_reached',
   593→          limit: usage.limit,
   594→          used: usage.used,
   595→          planName: usage.planName,
   596→        },
   597→        { status: 403 }
   598→      );
   599→    }
   600→
   601→    // 6. Build the AI prompt
   602→    const { systemPrompt, userMessage } = buildPrompt(section, action, body);
   603→
   604→    // 7. Call the AI SDK
   605→    const zai = await ZAI.create();
   606→    const completion = await zai.chat.completions.create({
   607→      messages: [
   608→        { role: 'assistant', content: systemPrompt },
   609→        { role: 'user', content: userMessage },
   610→      ],
   611→      thinking: { type: 'disabled' },
   612→    });
   613→
   614→    const aiResponse = completion.choices[0]?.message?.content?.trim();
   615→
   616→    if (!aiResponse) {
   617→      return NextResponse.json(
   618→        { success: false, error: 'Failed to generate suggestion' },
   619→        { status: 500 }
   620→      );
   621→    }
   622→
   623→    // 8. Save to AiHistory
   624→    await db.aiHistory.create({
   625→      data: {
   626→        resume_id: id,
   627→        user_id: auth.userId,
   628→        section,
   629→        action,
   630→        prompt: systemPrompt + '\n\n' + userMessage,
   631→        result: aiResponse,
   632→        original_text: content || '',
   633→      },
   634→    });
   635→
   636→    // 9. Return success
   637→    return NextResponse.json({
   638→      success: true,
   639→      result: aiResponse,
   640→      action,
   641→      section,
   642→      usage: {
   643→        limit: usage.limit,
   644→        used: usage.used + 1, // reflect the just-incremented count
   645→      },
   646→    });
   647→  } catch (error) {
   648→    console.error('[AI Assist] Error:', error);
   649→
   650→    // Handle ZAI-specific errors gracefully
   651→    if (error instanceof Error) {
   652→      const message = error.message.toLowerCase();
   653→      if (
   654→        message.includes('rate limit') ||
   655→        message.includes('quota') ||
   656→        message.includes('too many requests')
   657→      ) {
   658→        return NextResponse.json(
   659→          { success: false, error: 'AI service is temporarily unavailable. Please try again in a moment.' },
   660→          { status: 503 }
   661→        );
   662→      }
   663→
   664→      if (
   665→        message.includes('api key') ||
   666→        message.includes('authentication') ||
   667→        message.includes('unauthorized')
   668→      ) {
   669→        return NextResponse.json(
   670→          { success: false, error: 'AI service configuration error. Please contact support.' },
   671→          { status: 500 }
   672→        );
   673→      }
   674→
   675→      if (
   676→        message.includes('timeout') ||
   677→        message.includes('network') ||
   678→        message.includes('fetch')
   679→      ) {
   680→        return NextResponse.json(
   681→          { success: false, error: 'AI service request timed out. Please try again.' },
   682→          { status: 504 }
   683→        );
   684→      }
   685→    }
   686→
   687→    return NextResponse.json(
   688→      { success: false, error: 'Failed to generate suggestion' },
   689→      { status: 500 }
   690→    );
   691→  }
   692→}
