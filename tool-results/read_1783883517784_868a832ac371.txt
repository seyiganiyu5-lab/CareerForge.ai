   485→function estimateYearsOfExperience(workExperience?: ResumeContext['workExperience']): string {
   486→  if (!workExperience || workExperience.length === 0) return 'several';
   487→  let totalYears = 0;
   488→  for (const exp of workExperience) {
   489→    if (exp.startDate) {
   490→      const start = new Date(exp.startDate);
   491→      const end = exp.current ? new Date() : (exp.endDate ? new Date(exp.endDate) : new Date());
   492→      const diffMs = end.getTime() - start.getTime();
   493→      totalYears += Math.max(0, Math.round(diffMs / (365.25 * 24 * 60 * 60 * 1000)));
   494→    }
   495→  }
   496→  return totalYears > 0 ? String(totalYears) : 'several';
   497→}

   526→
   527→export async function POST(
   528→  request: NextRequest,
   529→  { params }: { params: Promise<{ id: string }> }
   530→) {
   531→  try {
   532→    // 1. Authentication
   533→    const auth = await getAuthUser();
   534→    if (!auth) {
   535→      return NextResponse.json(
   536→        { success: false, error: 'Unauthorized' },
   537→        { status: 401 }
   538→      );
   539→    }
   540→
   541→    // 2. Parse & validate request body
   542→    const body: AiAssistBody = await request.json();
   543→    const { section, action, content, customPrompt, tone, resumeContext, itemId, field } = body;
   544→
   545→    if (!section || !action) {
   546→      return NextResponse.json(
   547→        { success: false, error: 'Missing required fields: section and action' },
   548→        { status: 400 }
   549→      );
   550→    }
   551→
   552→    if (!VALID_SECTIONS.includes(section as Section)) {
   553→      return NextResponse.json(
   554→        { success: false, error: `Invalid section. Must be one of: ${VALID_SECTIONS.join(', ')}` },
   555→        { status: 400 }
   556→      );
   557→    }
   558→
   559→    if (!VALID_ACTIONS.includes(action as Action)) {
   560→      return NextResponse.json(
   561→        { success: false, error: `Invalid action. Must be one of: ${VALID_ACTIONS.join(', ')}` },
   562→        { status: 400 }
   563→      );
   564→    }
   565→
   566→    if (action === 'custom' && !customPrompt) {
   567→      return NextResponse.json(
   568→        { success: false, error: 'customPrompt is required when action is "custom"' },
   569→        { status: 400 }
   570→      );
   571→    }
   572→
   573→    // 3. Resolve resume ID from route params
   574→    const { id } = await params;
   575→
   576→    // 4. Ownership check
   577→    const resume = await db.resume.findUnique({
   578→      where: { id },
   579→    });
   580→
   581→    if (!resume || resume.user_id !== auth.userId) {
   582→      return NextResponse.json(
   583→        { success: false, error: 'Resume not found' },
   584→        { status: 404 }
   585→      );
   586→    }
   587→
   588→    // 5. Feature gating
   589→    const usage = await checkAndIncrementUsage(auth.userId, FEATURE_KEYS.AI_RESUME_BUILDER);
   590→    if (!usage.allowed) {
   591→      return NextResponse.json(
   592→        {
   593→          success: false,
   594→          error: usage.reason || 'limit_reached',
   595→          limit: usage.limit,
   596→          used: usage.used,
   597→          planName: usage.planName,
   598→        },
   599→        { status: 403 }
   600→      );
   601→    }
   602→
   603→    // 6. Build the AI prompt
   604→    const { systemPrompt, userMessage } = buildPrompt(section, action, body);
   605→
   606→    // 7. Call the AI SDK
   607→    const zai = await ZAI.create();
   608→    const completion = await zai.chat.completions.create({
   609→      messages: [
   610→        { role: 'assistant', content: systemPrompt },
   611→        { role: 'user', content: userMessage },
   612→      ],
   613→      thinking: { type: 'disabled' },
   614→    });
   615→
   616→    const aiResponse = completion.choices[0]?.message?.content?.trim();
   617→
   618→    if (!aiResponse) {
   619→      return NextResponse.json(
   620→        { success: false, error: 'Failed to generate suggestion' },
   621→        { status: 500 }
   622→      );
   623→    }
   624→
   625→    // Sanitize: strip literal \u2022 and other unicode bullet escapes the AI
   626→    // might produce, replacing them with plain dashes for consistent formatting.
   627→    const cleanResult = aiResponse
   628→      .replace(/\\u2022/g, '-')
   629→      .replace(/\u2022/g, '-');
   630→
   631→    // 8. Save to AiHistory
   632→    await db.aiHistory.create({
   633→      data: {
   634→        resume_id: id,
   635→        user_id: auth.userId,
   636→        section,
   637→        action,
   638→        prompt: systemPrompt + '\n\n' + userMessage,
   639→        result: cleanResult,
   640→        original_text: content || '',
   641→      },
   642→    });
   643→
   644→    // 9. Return success
   645→    return NextResponse.json({
   646→      success: true,
   647→      result: cleanResult,
   648→      action,
   649→      section,
   650→      usage: {
   651→        limit: usage.limit,
   652→        used: usage.used + 1, // reflect the just-incremented count
   653→      },
   654→    });
   655→  } catch (error) {
   656→    console.error('[AI Assist] Error:', error);
   657→
   658→    // Handle ZAI-specific errors gracefully
   659→    if (error instanceof Error) {
   660→      const message = error.message.toLowerCase();
   661→      if (
   662→        message.includes('rate limit') ||
   663→        message.includes('quota') ||
   664→        message.includes('too many requests')
   665→      ) {
   666→        return NextResponse.json(
   667→          { success: false, error: 'AI service is temporarily unavailable. Please try again in a moment.' },
   668→          { status: 503 }
   669→        );
   670→      }
   671→
   672→      if (
   673→        message.includes('api key') ||
   674→        message.includes('authentication') ||
   675→        message.includes('unauthorized')
   676→      ) {
   677→        return NextResponse.json(
   678→          { success: false, error: 'AI service configuration error. Please contact support.' },
   679→          { status: 500 }
   680→        );
   681→      }
   682→
   683→      if (
   684→        message.includes('timeout') ||
   685→        message.includes('network') ||
   686→        message.includes('fetch')
   687→      ) {
   688→        return NextResponse.json(
   689→          { success: false, error: 'AI service request timed out. Please try again.' },
   690→          { status: 504 }
   691→        );
   692→      }
   693→    }
   694→
   695→    return NextResponse.json(
   696→      { success: false, error: 'Failed to generate suggestion' },
   697→      { status: 500 }
   698→    );
   699→  }
   700→}
