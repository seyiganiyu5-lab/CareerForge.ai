   142→
   143→const LOADING_MESSAGES = [
   144→  'Thinking...',
   145→  'Generating...',
   146→  'Improving...',
   147→  'Analyzing...',
   148→  'Crafting...',
   149→  'Polishing...',
   150→];
   151→
   152→// ---------------------------------------------------------------------------
   153→// Framer Motion variants

   155→

   158→  visible: { opacity: 1, y: 0, transition: { duration: 0.3, ease: 'easeOut' } },

   162→// ---------------------------------------------------------------------------
   163→// Component
   164→// ---------------------------------------------------------------------------
   165→
   166→export function AiAssistPanel({
   167→  open,
   168→  onOpenChange,
   169→  resumeId,
   170→  section,
   171→  content,
   172→  itemId,
   173→  field,
   174→  resumeData,
   175→  onApply,
   176→  onInsertBelow,
   177→}: AiAssistPanelProps) {
   178→  // ---- i18n ----
   179→  const t = useTranslations('ai.assistant');
   180→
   181→  // ---- State ----
   182→  const [results, setResults] = useState<ResultItem[]>([]);
   183→  const [customPrompt, setCustomPrompt] = useState('');
   184→  const [isSending, setIsSending] = useState(false);
   185→  const [loadingMessage, setLoadingMessage] = useState(LOADING_MESSAGES[0]);
   186→  const scrollRef = useRef<HTMLDivElement>(null);
   187→  const requestInProgressRef = useRef(false);
   188→
   189→  const actions = SECTION_ACTIONS[section] ?? [
   190→    { action: 'improve', icon: Lightbulb, label: 'Improve' },
   191→    { action: 'rewrite', icon: RotateCcw, label: 'Rewrite' },
   192→    { action: 'grammar', icon: ShieldCheck, label: 'Grammar Check' },
   193→  ];
   194→
   195→  const sectionDisplayName = SECTION_DISPLAY_NAMES[section] ?? section;
   196→
   197→  // ---- Loading message cycling ----
   198→  useEffect(() => {
   199→    if (!isSending) return;
   200→    let index = 0;
   201→    const interval = setInterval(() => {
   202→      index = (index + 1) % LOADING_MESSAGES.length;
   203→      setLoadingMessage(LOADING_MESSAGES[index]);
   204→    }, 2000);
   205→    return () => clearInterval(interval);
   206→  }, [isSending]);
   207→
   208→  // ---- Clear state when panel closes ----
   209→  const handleOpenChange = useCallback(
   210→    (nextOpen: boolean) => {
   211→      if (!nextOpen) {
   212→        setResults([]);
   213→        setCustomPrompt('');
   214→        setIsSending(false);
   215→        setLoadingMessage(LOADING_MESSAGES[0]);
   216→        requestInProgressRef.current = false;
   217→      }
   218→      onOpenChange(nextOpen);
   219→    },
   220→    [onOpenChange],
   221→  );
   222→
   223→  // ---- Send AI request ----
   224→  const sendAiRequest = useCallback(
   225→    async (action: string, customPromptText?: string) => {
   226→      if (requestInProgressRef.current) return;
   227→      requestInProgressRef.current = true;
   228→      setIsSending(true);
   229→
   230→      const resultId = Date.now().toString(36);
   231→      setResults((prev) => [
   232→        ...prev,
   233→        { id: resultId, text: '', action, loading: true, copied: false },
   234→      ]);
   235→
   236→      // Scroll to bottom after the new result is rendered
   237→      setTimeout(() => {
   238→        scrollRef.current?.scrollTo({
   239→          top: scrollRef.current.scrollHeight,
   240→          behavior: 'smooth',
   241→        });
   242→      }, 100);
   243→
   244→      try {
   245→        const res = await fetch(`/api/resumes/${resumeId}/ai-assist`, {
   246→          method: 'POST',
   247→          headers: { 'Content-Type': 'application/json' },
   248→          credentials: 'include',
   249→          body: JSON.stringify({
   250→            section,
   251→            action,
   252→            content,
   253→            itemId,
   254→            field,
   255→            customPrompt: customPromptText,
   256→            resumeContext: resumeData,
   257→          }),
   258→        });
   259→
   260→        const data = await res.json();
   261→
   262→        if (!res.ok) {
   263→          if (data.error === 'limit_reached' || data.error === 'upgrade_required') {
   264→            setResults((prev) => prev.filter((r) => r.id !== resultId));
   265→            onOpenChange(false);
   266→            window.dispatchEvent(
   267→              new CustomEvent('ai-limit-reached', { detail: data }),
   268→            );
   269→            return;
   270→          }
   271→          throw new Error(data.error || 'Failed to generate suggestion');
   272→        }
   273→
   274→        setResults((prev) =>
   275→          prev.map((r) =>
   276→            r.id === resultId
   277→              ? { ...r, text: data.result, loading: false }
   278→              : r,
   279→          ),
   280→        );
   281→      } catch (err) {
   282→        setResults((prev) =>
   283→          prev.map((r) =>
   284→            r.id === resultId
   285→              ? {
   286→                  ...r,
   287→                  loading: false,
   288→                  error:
   289→                    err instanceof Error
   290→                      ? err.message
   291→                      : 'An error occurred',
   292→                }
   293→              : r,
   294→          ),
   295→        );
   296→      } finally {
   297→        setIsSending(false);
   298→        requestInProgressRef.current = false;
   299→      }
   300→    },
   301→    [resumeId, section, content, itemId, field, resumeData, onOpenChange],
   302→  );
   303→
   304→  // ---- Copy to clipboard ----
   305→  const copyToClipboard = useCallback(
   306→    async (text: string, resultId: string) => {
   307→      await navigator.clipboard.writeText(text);
   308→      setResults((prev) =>
   309→        prev.map((r) => (r.id === resultId ? { ...r, copied: true } : r)),
   310→      );
   311→      setTimeout(() => {
   312→        setResults((prev) =>
   313→          prev.map((r) =>
   314→            r.id === resultId ? { ...r, copied: false } : r,
   315→          ),
   316→        );
   317→      }, 2000);
   318→    },
   319→    [],
   320→  );
   321→
   322→  // ---- Apply result ----
   323→  const handleApply = useCallback(
   324→    (text: string) => {
   325→      onApply(text);
   326→      onOpenChange(false);
   327→    },
   328→    [onApply, onOpenChange],
   329→  );
   330→
   331→  // ---- Handle custom prompt submit ----
   332→  const handleCustomSubmit = useCallback(() => {
   333→    const trimmed = customPrompt.trim();
   334→    if (!trimmed || requestInProgressRef.current) return;
   335→    sendAiRequest('custom', trimmed);
   336→    setCustomPrompt('');
   337→  }, [customPrompt, sendAiRequest]);
   338→
   339→  // ---- Handle action button click ----
   340→  const handleActionClick = useCallback(
   341→    (action: string) => {
   342→      if (requestInProgressRef.current) return;
   343→      sendAiRequest(action);
   344→    },
   345→    [sendAiRequest],
   346→  );
   347→
   348→  return (
   349→    <Sheet open={open} onOpenChange={handleOpenChange}>
   350→      <TooltipProvider delayDuration={300}>
   351→        <SheetContent
   352→          side="right"
   353→          className="sm:max-w-[480px] p-0 gap-0 [&>button.absolute]:hidden"
   354→        >
   355→          {/* =========================================================== */}
   356→          {/* Header                                                      */}
   357→          {/* =========================================================== */}
   358→          <div className="relative flex items-center justify-between border-b border-border px-5 py-4">
   359→            {/* Gradient accent line */}
   360→            <div className="absolute inset-x-0 top-0 h-[3px] bg-gradient-to-r from-brand via-violet to-emerald" />
   361→
   362→            <div className="flex items-center gap-3">
   363→              <div className="flex size-9 items-center justify-center rounded-lg bg-brand/10">
   364→                <Sparkles className="size-5 text-brand" />
   365→              </div>
   366→              <div>
   367→                <SheetTitle className="font-heading text-base font-semibold text-foreground">
   368→                  AI Resume Assistant
   369→                </SheetTitle>
   370→                <p className="mt-0.5 text-xs text-muted-foreground font-body">
   371→                  {t('subtitle') || 'Powered by AI'}
   372→                </p>
   373→              </div>
   374→            </div>
   375→
   376→            <Button
   377→              variant="ghost"
   378→              size="icon"
   379→              className="size-8 rounded-md"
   380→              onClick={() => handleOpenChange(false)}
   381→            >
   382→              <X className="size-4" />
   383→              <span className="sr-only">Close</span>
   384→            </Button>
   385→          </div>
   386→
   387→          {/* =========================================================== */}
   388→          {/* Scrollable body                                             */}
   389→          {/* =========================================================== */}
   390→          <div
   391→            ref={scrollRef}
   392→            className="flex-1 overflow-y-auto overscroll-contain"
   393→          >
   394→            {/* --- Section badge & context --- */}
   395→            <div className="px-5 pt-4 pb-2">
   396→              <div className="flex items-center gap-2">
   397→                <Badge
   398→                  variant="secondary"
   399→                  className="bg-brand/10 text-brand border-brand/20 font-body font-medium text-xs"
   400→                >
   401→                  {sectionDisplayName}
   402→                </Badge>
   403→                {content && (
   404→                  <Badge
   405→                    variant="outline"
   406→                    className="text-muted-foreground font-body text-[11px]"
   407→                  >
   408→                    {content.length > 50
   409→                      ? `${content.length} chars`
   410→                      : `"${content.slice(0, 50)}${content.length > 50 ? '...' : ''}"`}
   411→                  </Badge>
   412→                )}
   413→              </div>
   414→              <p className="mt-3 text-sm text-muted-foreground font-body leading-relaxed">
   415→                {t('description') ||
   416→                  'Select a quick action below or type a custom request to enhance this section.'}
   417→              </p>
   418→            </div>
   419→
   420→            <Separator className="mx-5 w-auto" />
   421→
   422→            {/* --- Quick actions grid --- */}
   423→            <div className="px-5 pt-4 pb-2">
   424→              <p className="mb-3 text-xs font-medium uppercase tracking-wider text-muted-foreground font-body">
   425→                {t('quickActions') || 'Quick Actions'}
   426→              </p>
   427→              <div className="grid grid-cols-2 gap-2">
   428→                <AnimatePresence mode="popLayout">
   429→                  {actions.map((item) => {
   430→                    const Icon = item.icon;
   431→                    return (
   432→                      <motion.button
   433→                        key={item.action}
   434→                        type="button"
   435→                        whileHover={{ scale: 1.03 }}
   436→                        whileTap={{ scale: 0.97 }}
   437→                        onClick={() => handleActionClick(item.action)}
   438→                        disabled={isSending}
   439→                        className="group flex items-center gap-2.5 rounded-lg border border-border bg-card px-3 py-2.5 text-left text-sm font-medium text-foreground transition-colors hover:border-brand/40 hover:bg-brand/5 disabled:pointer-events-none disabled:opacity-50 font-body"
   440→                      >
   441→                        <Icon className="size-4 shrink-0 text-muted-foreground transition-colors group-hover:text-brand" />
   442→                        <span className="truncate">{item.label}</span>
   443→                      </motion.button>
   444→                    );
   445→                  })}
   446→                </AnimatePresence>
   447→              </div>
   448→            </div>
   449→
   450→            <Separator className="mx-5 w-auto" />
   451→
   452→            {/* --- Custom prompt --- */}
   453→            <div className="px-5 pt-4 pb-2">
   454→              <p className="mb-3 text-xs font-medium uppercase tracking-wider text-muted-foreground font-body">
   455→                {t('customPrompt') || 'Or ask AI anything'}
   456→              </p>
   457→              <div className="relative">
   458→                <Textarea
   459→                  value={customPrompt}
   460→                  onChange={(e) => setCustomPrompt(e.target.value)}
   461→                  placeholder={
   462→                    t('placeholder') ||
   463→                    'Type your request... e.g., "Make it more concise and action-oriented"'
   464→                  }
   465→                  rows={3}
   466→                  className="resize-none rounded-lg border-border bg-card pr-12 text-sm font-body placeholder:text-muted-foreground/60 focus-visible:ring-brand/30"
   467→                  onKeyDown={(e) => {
   468→                    if (e.key === 'Enter' && (e.metaKey || e.ctrlKey)) {
   469→                      e.preventDefault();
   470→                      handleCustomSubmit();
   471→                    }
   472→                  }}
   473→                />
   474→                <Tooltip>
   475→                  <TooltipTrigger asChild>
   476→                    <Button
   477→                      size="icon"
   478→                      onClick={handleCustomSubmit}
   479→                      disabled={!customPrompt.trim() || isSending}
   480→                      className="absolute bottom-2 right-2 size-8 rounded-md bg-brand text-primary-foreground shadow-sm hover:bg-brand/90 disabled:opacity-40"
   481→                    >
   482→                      {isSending ? (
   483→                        <Loader2 className="size-4 animate-spin" />
   484→                      ) : (
   485→                        <Send className="size-4" />
   486→                      )}
   487→                      <span className="sr-only">Send</span>
   488→                    </Button>
   489→                  </TooltipTrigger>
   490→                  <TooltipContent side="left" className="text-xs">
   491→                    {t('sendTooltip') || 'Send (⌘+Enter)'}
   492→                  </TooltipContent>
   493→                </Tooltip>
   494→              </div>
   495→            </div>
   496→
   497→            <Separator className="mx-5 w-auto" />
   498→
   499→            {/* --- Results --- */}
   500→            <div className="px-5 pt-4 pb-6">
   501→              <p className="mb-3 text-xs font-medium uppercase tracking-wider text-muted-foreground font-body">
   502→                {t('results') || 'Results'}
   503→              </p>
   504→
   505→              <AnimatePresence mode="popLayout">
   506→                {results.length === 0 && (
   507→                  <motion.div
   508→                    initial={{ opacity: 0 }}
   509→                    animate={{ opacity: 1 }}
   510→                    exit={{ opacity: 0 }}
   511→                    className="flex flex-col items-center justify-center rounded-xl border border-dashed border-border py-10 text-center"
   512→                  >
   513→                    <div className="mb-3 flex size-12 items-center justify-center rounded-full bg-brand/10">
   514→                      <Sparkles className="size-5 text-brand" />
   515→                    </div>
   516→                    <p className="text-sm font-medium text-muted-foreground font-body">
   517→                      {t('noResults') || 'No suggestions yet'}
   518→                    </p>
   519→                    <p className="mt-1 max-w-[200px] text-xs text-muted-foreground/70 font-body">
   520→                      {t('noResultsHint') ||
   521→                        'Click a quick action or type a request to get started'}
   522→                    </p>
   523→                  </motion.div>
   524→                )}
   525→
   526→                {results.map((result) => (
   527→                  <motion.div
   528→                    key={result.id}
   529→                    variants={resultVariants}
   530→                    initial="hidden"
   531→                    animate="visible"
   532→                    exit="exit"
   533→                    layout
   534→                    className="mb-4 last:mb-0"
   535→                  >
   536→                    {/* ---- Loading state ---- */}
   537→                    {result.loading && (
   538→                      <div className="rounded-xl border border-brand/20 bg-brand/5 p-4">
   539→                        <div className="mb-3 flex items-center gap-2">
   540→                          <Loader2 className="size-4 animate-spin text-brand" />
   541→                          <span className="text-sm font-medium text-brand font-body">
   542→                            {loadingMessage}
   543→                          </span>
   544→                        </div>
   545→                        <div className="space-y-2.5">
   546→                          <Skeleton className="h-4 w-full rounded-md" />
   547→                          <Skeleton className="h-4 w-11/12 rounded-md" />
   548→                          <Skeleton className="h-4 w-4/5 rounded-md" />
   549→                          <Skeleton className="h-4 w-9/12 rounded-md" />
   550→                        </div>
   551→                      </div>
   552→                    )}
   553→
   554→                    {/* ---- Error state ---- */}
   555→                    {!result.loading && result.error && (
   556→                      <div className="rounded-xl border border-destructive/30 bg-destructive/5 p-4">
   557→                        <p className="mb-3 text-sm font-medium text-destructive font-body">
   558→                          {result.error}
   559→                        </p>
   560→                        <Button
   561→                          variant="outline"
   562→                          size="sm"
   563→                          onClick={() => sendAiRequest(result.action)}
   564→                          className="text-destructive border-destructive/30 hover:bg-destructive/10 hover:text-destructive font-body"
   565→                        >
   566→                          <RotateCcw className="size-3.5" />
   567→                          {t('retry') || 'Retry'}
   568→                        </Button>
   569→                      </div>
   570→                    )}
   571→
   572→                    {/* ---- Success state ---- */}
   573→                    {!result.loading && !result.error && result.text && (
   574→                      <div className="overflow-hidden rounded-xl border border-border bg-card shadow-sm transition-shadow hover:shadow-md">
   575→                        {/* Left accent border */}
   576→                        <div className="flex">
   577→                          <div className="w-1 shrink-0 bg-gradient-to-b from-brand to-violet" />
   578→
   579→                          <div className="flex flex-1 flex-col">
   580→                            {/* Result text */}
   581→                            <div className="px-4 pt-4 pb-3">
   582→                              <p className="whitespace-pre-wrap text-sm leading-relaxed text-foreground font-body">
   583→                                {result.text}
   584→                              </p>
   585→                            </div>
   586→
   587→                            <Separator />
   588→
   589→                            {/* Action buttons */}
   590→                            <div className="flex flex-wrap items-center gap-1.5 px-4 py-3">
   591→                              {/* Apply — primary brand */}
   592→                              <Tooltip>
   593→                                <TooltipTrigger asChild>
   594→                                  <Button
