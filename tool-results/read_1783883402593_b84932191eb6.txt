   137→   137→      section: 'personalInfo',
   138→   138→    });

   141→   141→    score -= 20;
   142→   142→    suggestions.push({
   143→   143→      category: 'contactInfo',

   191→   191→function analyzeFormatting(resume: ResumeData): { score: number; issues: FormattingIssue[] } {
   192→   192→  let score = 100;
   193→   193→  const issues: FormattingIssue[] = [];
   194→   194→
   195→   195→  // Check professional summary
   196→   196→  if (!resume.professionalSummary || resume.professionalSummary.trim().length === 0) {
   197→   197→    score -= 20;
   198→   198→    issues.push({
   199→   199→      issue: 'Missing professional summary',
   200→   200→      suggestion: 'Add a 2-3 sentence professional summary highlighting your key achievements and target role',
   201→   201→      severity: 'error',
   202→   202→    });
   203→   203→  } else if (resume.professionalSummary.trim().length <= 20) {
   204→   204→    score -= 12;
   205→   205→    issues.push({
   206→   206→      issue: 'Professional summary is too short',
   207→   207→      suggestion: 'Expand your summary to 2-3 sentences highlighting key achievements and target role',
   208→   208→      severity: 'warning',
   209→   209→    });
   210→   210→  }
   211→   211→
   212→   212→  // Check work experience
   213→   213→  if (!resume.workExperience || resume.workExperience.length === 0) {
   214→   214→    score -= 25;
   215→   215→    issues.push({
   216→   216→      issue: 'No work experience entries',
   217→   217→      suggestion: 'Add at least one work experience entry with job title, company, and description',
   218→   218→      severity: 'error',
   219→   219→    });
   220→   220→  } else {
   221→   221→    const hasDescription = resume.workExperience.some(
   222→   222→      (exp) => exp.description && exp.description.trim().length > 0
   223→   223→    );
   224→   224→    if (!hasDescription) {
   225→   225→      score -= 15;
   226→   226→      issues.push({
   227→   227→        issue: 'Work experience entries lack descriptions',
   228→   228→        suggestion: 'Add bullet-point descriptions for each role highlighting responsibilities and achievements',
   229→   229→        severity: 'error',
   230→   230→      });
   231→   231→    }
   232→   232→  }
   233→   233→
   234→   234→  // Check education
   235→   235→  if (!resume.education || resume.education.length === 0) {
   236→   236→    score -= 15;
   237→   237→    issues.push({
   238→   238→      issue: 'No education entries',
   239→   239→      suggestion: 'Add at least one education entry with degree, field of study, and school',
   240→   240→      severity: 'warning',
   241→   241→    });
   242→   242→  }
   243→   243→
   244→   244→  // Check skills
   245→   245→  if (!resume.skills || resume.skills.length === 0) {
   246→   246→    score -= 15;
   247→   247→    issues.push({
   248→   248→      issue: 'No skills listed',
   249→   249→      suggestion: 'Add at least 3 relevant skills to help ATS systems match your resume',
   250→   250→      severity: 'error',
   251→   251→    });
   252→   252→  } else if (resume.skills.length < 3) {
   253→   253→    score -= 8;
   254→   254→    issues.push({
   255→   255→      issue: 'Too few skills listed',
   256→   256→      suggestion: 'Add more skills — aim for at least 7-10 relevant skills for better ATS matching',
   257→   257→      severity: 'warning',
   258→   258→    });
   259→   259→  }
   260→   260→
   261→   261→  // Check date consistency in work experience
   262→   262→  if (resume.workExperience && resume.workExperience.length > 0) {
   263→   263→    // Recognized date formats: "Jan 2023", "January 2023", "01/2023", "01/01/2023", "2023", "2024-12" (YYYY-MM)
   264→   264→    const datePatterns = [
   265→   265→      /^(jan|feb|mar|apr|may|jun|jul|aug|sep|oct|nov|dec|january|february|march|april|may|june|july|august|september|october|november|december)\s*\d{4}$/i,
   266→   266→      /^\d{1,2}\/\d{4}$/,
   267→   267→      /^\d{1,2}\/\d{1,2}\/\d{2,4}$/,
   268→   268→      /^\d{4}$/,
   269→   269→      /^\d{4}-\d{1,2}$/,  // YYYY-MM format (e.g., 2024-12)
   270→   270→    ];
   271→   271→
   272→   272→    // First, determine which format is used by the first entry
   273→   273→    let detectedFormat: number | null = null;
   274→   274→    const dates = resume.workExperience
   275→   275→      .flatMap((exp) => [exp.startDate, exp.endDate, exp.current ? 'Present' : null])
   276→   276→      .filter((d): d is string => !!d && d.trim() !== '' && d.trim().toLowerCase() !== 'present');
   277→   277→
   278→   278→    for (const date of dates) {
   279→   279→      for (let i = 0; i < datePatterns.length; i++) {
   280→   280→        if (datePatterns[i].test(date.trim())) {
   281→   281→          if (detectedFormat === null) {
   282→   282→            detectedFormat = i;
   283→   283→          } else if (detectedFormat !== i) {
   284→   284→            // Inconsistent: different format detected
   285→   285→            score -= 3;
   286→   286→            issues.push({
   287→   287→              issue: 'Mixed date formats detected across work experience entries',
   288→   288→              suggestion: 'Use a consistent date format like "Jan 2023", "01/2023", or "2024-12" throughout',
   289→   289→              severity: 'warning',
   290→   290→            });
   291→   291→            break;
   292→   292→          }
   293→   293→          break;
   294→   294→        }
   295→   295→      }
   296→   296→      // If a date doesn't match ANY known format, flag it
   297→   297→      if (detectedFormat !== null) break; // already flagged inconsistency
   298→   298→      const matchesAny = datePatterns.some((p) => p.test(date.trim()));
   299→   299→      if (!matchesAny) {
   300→   300→        score -= 3;
   301→   301→        issues.push({
   302→   302→          issue: `Unrecognized date format: "${date}"`,
   303→   303→          suggestion: 'Use a standard format like "Jan 2023", "MM/YYYY", or "YYYY-MM"',
   304→   304→          severity: 'warning',
   305→   305→        });
   306→   306→        break;
   307→   307→      }
   308→   308→    }
   309→   309→  }
   310→   310→
   311→   311→  return { score: Math.max(0, Math.min(100, score)), issues };
   312→   312→}

   314→   314→function analyzeWorkExperience(workExperience: WorkExperience[]): { score: number; suggestions: Suggestion[] } {
   315→   315→  if (!workExperience || workExperience.length === 0) {
   316→   316→    return {
   317→   317→      score: 0,
   318→   318→      suggestions: [
   319→   319→        {
   320→   320→          category: 'workExperience',
   321→   321→          message: 'Add work experience to significantly improve your resume score',
   322→   322→          priority: 'high',
   323→   323→          section: 'workExperience',
   324→   324→        },
   325→   325→      ],
   326→   326→    };
   327→   327→  }
   328→   328→
   329→   329→  let score = 0;
   330→   330→  const suggestions: Suggestion[] = [];
   331→   331→
   332→   332→  // Number of entries scoring
   333→   333→  const count = workExperience.length;
   334→   334→  if (count === 1) score = 40;
   335→   335→  else if (count === 2) score = 70;
   336→   336→  else if (count >= 3) score = 90;
   337→   337→
   338→   338→  // Average description length
   339→   339→  const entriesWithDesc = workExperience.filter((e) => e.description && e.description.trim().length > 0);
   340→   340→  if (entriesWithDesc.length > 0) {
   341→   341→    const avgDescLen = entriesWithDesc.reduce((sum, e) => sum + e.description.trim().length, 0) / entriesWithDesc.length;
   342→   342→
   343→   343→    if (avgDescLen < 50) {
   344→   344→      score -= 20;
   345→   345→      suggestions.push({
   346→   346→        category: 'workExperience',
   347→   347→        message: 'Work experience descriptions are too brief — add more detail about your responsibilities and achievements',
   348→   348→        priority: 'high',
   349→   349→        section: 'workExperience',
   350→   350→      });
   351→   351→    } else if (avgDescLen < 150) {
   352→   352→      score -= 10;
   353→   353→      suggestions.push({
   354→   354→        category: 'workExperience',
   355→   355→        message: 'Expand your work experience descriptions with more specific details and accomplishments',
   356→   356→        priority: 'medium',
   357→   357→        section: 'workExperience',
   358→   358→      });
   359→   359→    } else if (avgDescLen >= 150) {
   360→   360→      score += 5; // Bonus for detailed descriptions
   361→   361→    }
   362→   362→  } else {
   363→   363→    score -= 20;
   364→   364→    suggestions.push({
   365→   365→      category: 'workExperience',
   366→   366→      message: 'Add descriptions to your work experience entries with bullet points',
   367→   367→      priority: 'high',
   368→   368→      section: 'workExperience',
   369→   369→    });
   370→   370→  }
   371→   371→
   372→   372→  // Current job flag
   373→   373→  const hasCurrentJob = workExperience.some((e) => e.currentJob);
   374→   374→  if (hasCurrentJob) {
   375→   375→    score += 5;
   376→   376→  }
   377→   377→
   378→   378→  // Action verbs check
   379→   379→  let totalBullets = 0;
   380→   380→  let actionVerbBullets = 0;
   381→   381→
   382→   382→  for (const exp of entriesWithDesc) {
   383→   383→    const bullets = exp.description
   384→   384→      .split('\n')
   385→   385→      .map((b) => b.trim())
   386→   386→      .filter((b) => b.length > 0 && (b.startsWith('-') || b.startsWith('•') || b.startsWith('*')));
   387→   387→
   388→   388→    for (const bullet of bullets) {
   389→   389→      const cleanBullet = bullet.replace(/^[-•*]\s*/, '').trim();
   390→   390→      if (cleanBullet.length === 0) continue;
   391→   391→      totalBullets++;
   392→   392→      const firstWord = cleanBullet.split(/\s+/)[0];
   393→   393→      if (ACTION_VERBS.some((verb) => firstWord.toLowerCase() === verb.toLowerCase())) {
   394→   394→        actionVerbBullets++;
   395→   395→      }
   396→   396→    }
   397→   397→  }
   398→   398→
   399→   399→  if (totalBullets > 0) {
   400→   400→    const actionVerbRatio = actionVerbBullets / totalBullets;
   401→   401→    if (actionVerbRatio < 0.3) {
   402→   402→      score -= 10;
   403→   403→      suggestions.push({
   404→   404→        category: 'workExperience',
   405→   405→        message: 'Start more bullet points with strong action verbs (Led, Developed, Achieved, Implemented, etc.)',
   406→   406→        priority: 'medium',
   407→   407→        section: 'workExperience',
   408→   408→      });
   409→   409→    } else if (actionVerbRatio >= 0.6) {
   410→   410→      score += 5; // Bonus for strong action verbs
   411→   411→    }
   412→   412→  }
   413→   413→
   414→   414→  return { score: Math.max(0, Math.min(100, score)), suggestions };
   415→   415→}

   530→   530→function analyzeReadability(resume: ResumeData): number {
   531→   531→  // Total word count across all text sections
   532→   532→  const texts: string[] = [];
   533→   533→
   534→   534→  if (resume.professionalSummary) texts.push(resume.professionalSummary);

   537→   537→  if (resume.workExperience) {
   538→   538→    for (const exp of resume.workExperience) {
   539→   539→      if (exp.description) texts.push(exp.description);
   540→   540→      if (exp.position) texts.push(exp.position);
   541→   541→    }
   542→   542→  }
   543→   543→
   544→   544→  if (resume.education) {
   545→   545→    for (const edu of resume.education) {

   547→   547→      if (edu.fieldOfStudy) texts.push(edu.fieldOfStudy);
   548→   548→    }
   549→   549→  }
   550→   550→
   551→   551→  if (resume.projects) {
   552→   552→    for (const proj of resume.projects) {

   557→   557→  if (resume.additionalInfo?.text) texts.push(resume.additionalInfo.text);

   560→   560→
   561→   561→  // Ideal range: 300-800 words
   562→   562→  if (totalWords < 100) return 20;
   563→   563→  if (totalWords < 200) return 45;

   565→   565→  if (totalWords <= 800) return 90;
   566→   566→  if (totalWords <= 1000) return 80;

   568→   568→  return 45; // Way too long

   571→   571→function analyzeAchievements(workExperience: WorkExperience[]): { score: number; suggestions: Suggestion[] } {
   572→   572→  if (!workExperience || workExperience.length === 0) {
   573→   573→    return { score: 0, suggestions: [] };
   574→   574→  }
   575→   575→
   576→   576→  const suggestions: Suggestion[] = [];
   577→   577→  let totalBullets = 0;
   578→   578→  let quantifiableBullets = 0;
   579→   579→
   580→   580→  const metricsPattern = /\d+%|\$[\d,.]+|\d+\+?\s*(?:users?|clients?|customers?|members?|people|projects?|teams?|reports?|sites?|pages?|articles?|products?|services?|applications?|systems?|processes?|campaigns?|accounts?)/i;
   581→   581→  const numberPattern = /\b\d{2,}\b/;
   582→   582→
   583→   583→  for (const exp of workExperience) {
   584→   584→    if (!exp.description) continue;
   585→   585→    const bullets = exp.description
   586→   586→      .split('\n')
   587→   587→      .map((b) => b.trim())
   588→   588→      .filter((b) => b.length > 0);
   589→   589→
   590→   590→    for (const bullet of bullets) {
   591→   591→      const clean = bullet.replace(/^[-•*]\s*/, '').trim();
   592→   592→      if (clean.length === 0) continue;
   593→   593→      totalBullets++;
   594→   594→
   595→   595→      if (metricsPattern.test(clean) || numberPattern.test(clean)) {
   596→   596→        quantifiableBullets++;
   597→   597→      }
   598→   598→    }
   599→   599→  }
   600→   600→
   601→   601→  if (totalBullets === 0) return { score: 10, suggestions };
   602→   602→
   603→   603→  const ratio = quantifiableBullets / totalBullets;
   604→   604→  let score: number;
   605→   605→
   606→   606→  if (ratio >= 0.6) score = 95;
   607→   607→  else if (ratio >= 0.4) score = 80;
   608→   608→  else if (ratio >= 0.2) score = 60;
   609→   609→  else if (ratio >= 0.1) score = 40;
   610→   610→  else score = 20;
   611→   611→
   612→   612→  if (ratio < 0.3) {
   613→   613→    suggestions.push({
   614→   614→      category: 'achievements',
   615→   615→      message: 'Add quantifiable achievements with numbers, percentages, or dollar amounts to your bullet points',
   616→   616→      priority: 'high',
   617→   617→      section: 'workExperience',
   618→   618→    });
   619→   619→  } else if (ratio < 0.5) {
   620→   620→    suggestions.push({
   621→   621→      category: 'achievements',
   622→   622→      message: 'More of your bullet points should include quantifiable results (e.g., "Increased sales by 25%")',
   623→   623→      priority: 'medium',
   624→   624→      section: 'workExperience',
   625→   625→    });
   626→   626→  }
   627→   627→
   628→   628→  return { score, suggestions };
   629→   629→}
