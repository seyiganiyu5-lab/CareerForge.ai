    99→// ── Helpers ──────────────────────────────────────────────────────────────────

   101→const PLAN_ORDER = ['free', 'pro', 'enterprise'] as const;
   102→type PlanName = (typeof PLAN_ORDER)[number];
   103→
   104→function getPlanTier(planName: string): number {
   105→  const idx = PLAN_ORDER.indexOf(planName.toLowerCase() as PlanName);
   106→  return idx >= 0 ? idx : 0;
   107→}
   108→
   109→function formatPrice(cents: number, currency: string = 'USD'): string {
   110→  const amount = cents / 100;
   111→  return new Intl.NumberFormat('en-US', {
   112→    style: 'currency',
   113→    currency,
   114→    minimumFractionDigits: 2,
   115→    maximumFractionDigits: 2,
   116→  }).format(amount);
   117→}
   118→
   119→function formatDate(dateStr: string | null): string {
   120→  if (!dateStr) return 'N/A';
   121→  return new Date(dateStr).toLocaleDateString('en-US', {
   122→    year: 'numeric',
   123→    month: 'long',
   124→    day: 'numeric',
   125→  });
   126→}
   127→
   128→function formatShortDate(dateStr: string | null): string {
   129→  if (!dateStr) return '—';
   130→  return new Date(dateStr).toLocaleDateString('en-US', {
   131→    year: 'numeric',
   132→    month: 'short',
   133→    day: 'numeric',
   134→  });
   135→}
   136→
   137→function getFeatureLabel(key: string): string {
   138→  return FEATURE_DISPLAY_INFO[key]?.label ?? key.replace(/_/g, ' ');
   139→}
   140→
   141→function getFeatureLimitText(feature: PlanFeature): string {
   142→  if (!feature.enabled) return '';
   143→  if (feature.limit === null || feature.limit === -1 || feature.limit === 0) return 'Unlimited';
   144→  return String(feature.limit);
   145→}

   565→              }`}
   566→            >
   567→              Monthly
   568→            </span>
   569→            <Switch
   570→              checked={isYearly}
   571→              onCheckedChange={setIsYearly}
   572→              aria-label="Toggle billing period"
   573→            />
   574→            <span
   575→              className={`text-sm font-medium font-[family-name:var(--font-inter)] transition-colors ${
   576→                isYearly ? 'text-foreground' : 'text-muted-foreground'

   578→            >
   579→              Yearly
   580→            </span>
   581→            <Badge className="bg-emerald/10 text-emerald border-emerald/20 text-xs font-[family-name:var(--font-inter)]">

   583→            </Badge>
   584→          </div>
   585→        </div>
   586→
   587→        <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-6">
   588→          {plans.map(plan => {

   590→            const tier = getPlanTier(planName);
   591→            const isCurrentPlan = planName === currentPlanName;
   592→            const isHigher = tier > currentTier;
   593→            const isPro = planName === 'pro';
   594→            const isEnterprise = planName === 'enterprise';
   595→            const price = isYearly ? plan.price_yearly : plan.price_monthly;
   596→            const otherPrice = isYearly ? plan.price_monthly : plan.price_yearly;

   599→            return (
   600→              <motion.div
   601→                key={plan.id}
   602→                variants={cardVariant}
   603→                className={`relative rounded-2xl border bg-card flex flex-col overflow-hidden transition-shadow hover:shadow-md ${
   604→                  isPro
   605→                    ? 'border-brand shadow-lg lg:scale-[1.02]'
   606→                    : isCurrentPlan
   607→                      ? 'border-brand/50'
   608→                      : 'border-border'
   609→                }`}
   610→              >
   611→                {/* Popular badge */}
   612→                {isPro && (
   613→                  <span className="absolute -top-3 left-1/2 -translate-x-1/2 bg-brand text-white text-xs font-medium px-3 py-1 rounded-full font-[family-name:var(--font-inter)] z-10">
   614→                    Most Popular
   615→                  </span>
   616→                )}
   617→
   618→                {/* Card header area */}
   619→                <div className="p-6 pb-0 flex-1">
   620→                  <div className="flex items-center justify-between">
   621→                    <div className="flex items-center gap-2">
   622→                      <span className={PLAN_COLORS[planName] || 'text-foreground'}>
   623→                        {PLAN_ICONS[planName] || <CreditCard className="w-5 h-5" />}
   624→                      </span>
   625→                      <h3 className="text-lg font-bold font-[family-name:var(--font-poppins)] text-foreground">
   626→                        {plan.display_name}
   627→                      </h3>
   628→                    </div>
   629→                    {isCurrentPlan && (
   630→                      <Badge
   631→                        variant="outline"
   632→                        className="border-brand/30 text-brand text-xs font-[family-name:var(--font-inter)]"

   634→                        Current Plan
   635→                      </Badge>
   636→                    )}
   637→                  </div>
   638→
   639→                  <p className="text-sm text-muted-foreground font-[family-name:var(--font-inter)] mt-1">
   640→                    {plan.description}
   641→                  </p>
   642→
   643→                  {/* Price */}
   644→                  <div className="mt-4 flex items-baseline gap-1">

   646→                      <span className="text-4xl font-bold font-[family-name:var(--font-poppins)] text-foreground">
   647→                        Free
   648→                      </span>
   649→                    ) : (
   650→                      <>
   651→                        <motion.span
   652→                          key={isYearly ? 'yearly-price' : 'monthly-price'}

   654→                          animate={{ opacity: 1, y: 0 }}
   655→                          transition={{ duration: 0.2 }}
   656→                          className="text-4xl font-bold font-[family-name:var(--font-poppins)] text-foreground"

   658→                          {formatPrice(price, plan.currency)}
   659→                        </motion.span>
   660→                        <motion.span
   661→                          key={isYearly ? 'yearly-period' : 'monthly-period'}
   662→                          initial={{ opacity: 0, y: -8 }}
   663→                          animate={{ opacity: 1, y: 0 }}
   664→                          transition={{ duration: 0.2 }}
   665→                          className="text-muted-foreground text-sm font-[family-name:var(--font-inter)]"

   668→                        </motion.span>
   669→                      </>
   670→                    )}
   671→                  </div>
   672→
   673→                  {/* Yearly savings note */}
   674→                  {!isFreePlan && (
   675→                    <motion.p
   676→                      key={isYearly ? 'savings-on' : 'savings-off'}
   677→                      initial={{ opacity: 0 }}
   678→                      animate={{ opacity: 1 }}
   679→                      className="text-xs text-muted-foreground font-[family-name:var(--font-inter)] mt-1"
   680→                    >
   681→                      {isYearly
   682→                        ? `That's ${formatPrice(Math.round(otherPrice / 12), plan.currency)}/mo — billed annually`
   683→                        : `Or ${formatPrice(plan.price_yearly, plan.currency)}/yr — save ${formatPrice(plan.price_monthly * 12 - plan.price_yearly, plan.currency)}`
   684→                      }
   685→                    </motion.p>
   686→                  )}
   687→
   688→                  <Separator className="my-5" />
   689→
   690→                  {/* Feature list */}
   691→                  <ul className="space-y-3">
   692→                    {plan.features
   693→                      .filter(f => FEATURE_DISPLAY_INFO[f.key])
   694→                      .map(feature => {
   695→                        const label = getFeatureLabel(feature.key);
   696→                        const limitText = getFeatureLimitText(feature);
   697→
   698→                        return (
   699→                          <li key={feature.key} className="flex items-start gap-2.5">
   700→                            {feature.enabled ? (
   701→                              <CheckCircle2 className="w-4 h-4 text-emerald mt-0.5 shrink-0" />
   702→                            ) : (
   703→                              <XCircle className="w-4 h-4 text-muted-foreground/40 mt-0.5 shrink-0" />
   704→                            )}
   705→                            <span
   706→                              className={`text-sm font-[family-name:var(--font-inter)] ${
   707→                                feature.enabled
   708→                                  ? 'text-foreground/90'
   709→                                  : 'text-muted-foreground/50 line-through'
   710→                              }`}
   711→                            >
   712→                              {label}
   713→                              {feature.enabled && limitText && limitText !== 'Unlimited' && (
   714→                                <span className="text-muted-foreground ml-1">({limitText})</span>
   715→                              )}
   716→                              {feature.enabled && limitText === 'Unlimited' && (
   717→                                <span className="text-emerald ml-1 text-xs">(Unlimited)</span>
   718→                              )}
   719→                            </span>
   720→                          </li>
   721→                        );
   722→                      })}
   723→                  </ul>
   724→                </div>
   725→
   726→                {/* CTA button */}
   727→                <div className="p-6 pt-2 mt-auto">
   728→                  {isCurrentPlan ? (
   729→                    <Button
   730→                      variant="outline"
   731→                      className="w-full rounded-lg font-[family-name:var(--font-inter)]"
   732→                      disabled
   733→                    >
   734→                      Current Plan
   735→                    </Button>
   736→                  ) : isEnterprise ? (
   737→                    <Button
   738→                      variant="outline"
   739→                      className="w-full rounded-lg font-[family-name:var(--font-inter)]"
   740→                      onClick={() => toast.info('Contact our sales team at sales@careerforge.ai')}
   741→                    >
   742→                      Contact Sales
   743→                      <ArrowRight className="w-4 h-4 ml-1" />
   744→                    </Button>
   745→                  ) : isHigher ? (
   746→                    <Button
   747→                      className={`w-full rounded-lg font-[family-name:var(--font-inter)] ${
   748→                        isPro
   749→                          ? 'bg-brand text-white hover:bg-brand/90'
   750→                          : 'bg-foreground text-background hover:bg-foreground/90'
   751→                      }`}
   752→                      onClick={() => handleUpgrade(plan.id, plan.display_name)}
   753→                      disabled={upgrading !== null}
   754→                    >
   755→                      {upgrading === plan.id ? (
   756→                        <Loader2 className="w-4 h-4 animate-spin" />
   757→                      ) : (
   758→                        <>
   759→                          Upgrade to {plan.display_name}
   760→                          <ArrowRight className="w-4 h-4 ml-1" />
   761→                        </>
   762→                      )}
   763→                    </Button>
   764→                  ) : (
   765→                    <Button
   766→                      variant="outline"
   767→                      className="w-full rounded-lg font-[family-name:var(--font-inter)] text-rose-500 border-rose-500/30 hover:bg-rose-500/5 hover:text-rose-600"
   768→                      onClick={() => setCancelDialogOpen(true)}
   769→                    >
   770→                      Downgrade
   771→                    </Button>
   772→                  )}
   773→                </div>
   774→              </motion.div>
   775→            );
   776→          })}
   777→        </div>
   778→      </motion.div>
   779→
   780→      {/* ── 3. Payment History ─────────────────────────────────────────────── */}
   781→      <motion.div variants={fadeInUp}>
   782→        <Card>
   783→          <CardHeader>
   784→            <div className="flex items-center gap-2">
   785→              <Receipt className="w-5 h-5 text-muted-foreground" />
   786→              <CardTitle className="font-[family-name:var(--font-poppins)]">
   787→                Payment History
   788→              </CardTitle>
   789→            </div>
   790→            <CardDescription className="font-[family-name:var(--font-inter)]">
   791→              View your past transactions and payment details.
   792→            </CardDescription>
   793→          </CardHeader>
   794→          <CardContent>
   795→            {payments.length === 0 ? (
   796→              <EmptyState
   797→                icon={Receipt}
   798→                title="No payment history"
   799→                description="When you upgrade your plan, your payment history will appear here."
   800→              />
   801→            ) : (
   802→              <div className="overflow-x-auto -mx-6 px-6">
   803→                <table className="w-full text-sm font-[family-name:var(--font-inter)]">
   804→                  <thead>
   805→                    <tr className="border-b border-border">
   806→                      <th className="text-left py-3 px-3 text-xs font-medium text-muted-foreground uppercase tracking-wider">
   807→                        Date
   808→                      </th>
   809→                      <th className="text-left py-3 px-3 text-xs font-medium text-muted-foreground uppercase tracking-wider">
   810→                        Plan
   811→                      </th>
   812→                      <th className="text-left py-3 px-3 text-xs font-medium text-muted-foreground uppercase tracking-wider">
   813→                        Amount
   814→                      </th>
   815→                      <th className="text-left py-3 px-3 text-xs font-medium text-muted-foreground uppercase tracking-wider">
   816→                        Status
   817→                      </th>
   818→                      <th className="text-left py-3 px-3 text-xs font-medium text-muted-foreground uppercase tracking-wider hidden md:table-cell">
   819→                        Method
   820→                      </th>
   821→                    </tr>
   822→                  </thead>
   823→                  <tbody>
   824→                    {payments.map(payment => (
   825→                      <tr
   826→                        key={payment.id}
   827→                        className="border-b border-border/50 last:border-0 hover:bg-muted/30 transition-colors"
   828→                      >
   829→                        <td className="py-3.5 px-3 text-foreground">
   830→                          {formatShortDate(payment.paid_at || payment.created_at)}
   831→                        </td>
   832→                        <td className="py-3.5 px-3 text-foreground font-medium">
   833→                          {payment.plan_name}
   834→                        </td>
   835→                        <td className="py-3.5 px-3 text-foreground">
   836→                          {formatPrice(payment.amount, payment.currency)}
   837→                        </td>
   838→                        <td className="py-3.5 px-3">
   839→                          <Badge
   840→                            variant={
   841→                              payment.status === 'successful'
   842→                                ? 'secondary'
   843→                                : payment.status === 'pending'
   844→                                  ? 'outline'
   845→                                  : 'destructive'

   848→                              payment.status === 'successful'
   849→                                ? 'bg-emerald/10 text-emerald border-emerald/20'
   850→                                : ''
   851→                            }
   852→                          >
   853→                            {payment.status === 'successful'
   854→                              ? 'Paid'
   855→                              : payment.status === 'pending'
   856→                                ? 'Pending'
   857→                                : payment.status === 'failed'
   858→                                  ? 'Failed'
   859→                                  : payment.status.charAt(0).toUpperCase() + payment.status.slice(1)
   860→                            }
   861→                          </Badge>
   862→                        </td>
   863→                        <td className="py-3.5 px-3 text-muted-foreground capitalize hidden md:table-cell">
   864→                          {payment.payment_method || '—'}
   865→                        </td>
   866→                      </tr>
   867→                    ))}
   868→                  </tbody>
   869→                </table>
   870→              </div>
   871→            )}
   872→          </CardContent>
   873→        </Card>
   874→      </motion.div>
   875→
   876→      {/* ── Cancel / Downgrade Dialog ──────────────────────────────────────── */}
   877→      <Dialog open={cancelDialogOpen} onOpenChange={setCancelDialogOpen}>
   878→        <DialogContent>
   879→          <DialogHeader>
   880→            <DialogTitle className="font-[family-name:var(--font-poppins)]">
   881→              Downgrade to Free Plan?
   882→            </DialogTitle>
   883→            <DialogDescription className="font-[family-name:var(--font-inter)]">
   884→              You are about to downgrade from{' '}
   885→              <span className="font-medium text-foreground">
   886→                {subscription?.plan.display_name || 'your current plan'}
   887→              </span>{' '}
   888→              to the Free plan. This action will take effect immediately and you will
   889→              lose access to all premium features.
   890→            </DialogDescription>
   891→          </DialogHeader>
   892→
   893→          <div className="rounded-lg border border-rose-500/20 bg-rose-500/5 p-4 space-y-2">
   894→            <div className="flex items-start gap-2">
   895→              <AlertTriangle className="w-4 h-4 text-rose-500 mt-0.5 shrink-0" />
   896→              <p className="text-sm text-rose-700 dark:text-rose-300 font-[family-name:var(--font-inter)]">
   897→                <strong>What you&apos;ll lose:</strong> AI-powered resume builder,
   898→                AI cover letter generator, ATS checker, job tracker, unlimited exports,

   900→              </p>
   901→            </div>
   902→          </div>
   903→
   904→          <DialogFooter>
   905→            <Button
   906→              variant="outline"
   907→              className="rounded-lg font-[family-name:var(--font-inter)]"
   908→              onClick={() => setCancelDialogOpen(false)}
   909→              disabled={cancelling}
   910→            >
   911→              Keep Current Plan
   912→            </Button>
   913→            <Button
   914→              variant="destructive"
   915→              className="rounded-lg font-[family-name:var(--font-inter)]"
   916→              onClick={handleCancel}
   917→              disabled={cancelling}
   918→            >
   919→              {cancelling ? (
   920→                <Loader2 className="w-4 h-4 animate-spin" />
   921→              ) : (
   922→                'Downgrade to Free'
