   212→// ─── Animated Score Circle ───────────────────────────────────────────────────
   213→
   214→function AnimatedScoreCircle({ score, size = 160 }: { score: number; size?: number }) {
   215→  const [displayScore, setDisplayScore] = useState(0);
   216→  const strokeWidth = 10;
   217→  const radius = (size - strokeWidth) / 2;
   218→  const circumference = 2 * Math.PI * radius;
   219→  const center = size / 2;
   220→  const color = getScoreTrackColor(score);
   221→
   222→  useEffect(() => {
   223→    const duration = 1200;
   224→    const startTime = Date.now();
   225→    const animate = () => {
   226→      const elapsed = Date.now() - startTime;
   227→      const progress = Math.min(elapsed / duration, 1);
   228→      const eased = 1 - Math.pow(1 - progress, 3);
   229→      setDisplayScore(Math.round(score * eased));
   230→      if (progress < 1) requestAnimationFrame(animate);
   231→    };
   232→    const raf = requestAnimationFrame(animate);
   233→    return () => cancelAnimationFrame(raf);
   234→  }, [score]);
   235→
   236→  return (
   237→    <div className="relative inline-flex items-center justify-center" role="img" aria-label={`ATS Score: ${score} out of 100`}>
   238→      <svg width={size} height={size} className="-rotate-90">
   239→        <circle
   240→          cx={center}
   241→          cy={center}
   242→          r={radius}
   243→          fill="none"
   244→          stroke="currentColor"
   245→          className="text-muted/30"
   246→          strokeWidth={strokeWidth}
   247→        />
   248→        <motion.circle
   249→          cx={center}
   250→          cy={center}
   251→          r={radius}
   252→          fill="none"
   253→          className={color}
   254→          strokeWidth={strokeWidth}
   255→          strokeLinecap="round"
   256→          strokeDasharray={circumference}
   257→          initial={{ strokeDashoffset: circumference }}
   258→          animate={{ strokeDashoffset: circumference - (circumference * score) / 100 }}
   259→          transition={{ duration: 1.2, ease: [0.25, 0.46, 0.45, 0.94] }}
   260→        />
   261→      </svg>
   262→      <div className="absolute flex flex-col items-center">
   263→        <span className={`font-[family-name:var(--font-poppins)] text-4xl font-bold tabular-nums ${getScoreColor(score)}`}>
   264→          {displayScore}
   265→        </span>
   266→        <span className="text-xs text-muted-foreground">/ 100</span>
   267→      </div>
   268→    </div>
   269→  );
   270→}

   272→// ─── Loading State ───────────────────────────────────────────────────────────
   273→
   274→function LoadingState() {
   275→  const [currentStep, setCurrentStep] = useState(0);
   276→
   277→  useEffect(() => {
   278→    const interval = setInterval(() => {
   279→      setCurrentStep((prev) => (prev + 1) % LOADING_STEPS.length);
   280→    }, 2200);
   281→    return () => clearInterval(interval);
   282→  }, []);
   283→
   284→  return (
   285→    <motion.div
   286→      initial={{ opacity: 0, y: 20 }}
   287→      animate={{ opacity: 1, y: 0 }}
   288→      exit={{ opacity: 0, y: -10 }}
   289→      transition={{ duration: 0.3 }}
   290→      className="flex flex-col items-center gap-8 py-12"
   291→      role="status"
   292→      aria-label="Analyzing resume"
   293→    >
   294→      {/* Pulsing score circle skeleton */}
   295→      <div className="relative flex items-center justify-center">
   296→        <motion.div
   297→          className="size-40 rounded-full border-4 border-muted"
   298→          animate={{ scale: [1, 1.04, 1] }}
   299→          transition={{ duration: 2, repeat: Infinity, ease: 'easeInOut' }}
   300→        />
   301→        <motion.div
   302→          className="absolute inset-2 rounded-full border-4 border-dashed border-brand/30"
   303→          animate={{ rotate: 360 }}
   304→          transition={{ duration: 3, repeat: Infinity, ease: 'linear' }}
   305→        />
   306→        <div className="absolute flex flex-col items-center">
   307→          <Sparkles className="size-6 text-brand animate-pulse" aria-hidden="true" />
   308→          <span className="mt-2 text-sm font-medium text-muted-foreground">Analyzing...</span>
   309→        </div>
   310→      </div>
   311→
   312→      {/* Step indicators */}
   313→      <div className="w-full max-w-md space-y-3">
   314→        {LOADING_STEPS.map((step, idx) => (
   315→          <motion.div
   316→            key={step}
   317→            initial={{ opacity: 0, x: -12 }}
   318→            animate={{ opacity: 1, x: 0 }}
   319→            transition={{ delay: idx * 0.15, duration: 0.3 }}
   320→            className={`flex items-center gap-3 rounded-lg px-4 py-3 transition-colors duration-300 ${
   321→              idx === currentStep ? 'bg-brand/10 border border-brand/20' : 'border border-transparent'
   322→            }`}
   323→          >
   324→            <motion.div
   325→              animate={{
   326→                scale: idx === currentStep ? [1, 1.2, 1] : 1,
   327→                opacity: idx <= currentStep ? 1 : 0.3,
   328→              }}
   329→              transition={{ duration: 1, repeat: idx === currentStep ? Infinity : 0 }}
   330→            >
   331→              {idx < currentStep ? (
   332→                <CheckCircle2 className="size-5 text-emerald-500" aria-hidden="true" />
   333→              ) : (
   334→                <div className={`size-5 rounded-full border-2 ${idx === currentStep ? 'border-brand' : 'border-muted-foreground/30'}`} />
   335→              )}
   336→            </motion.div>
   337→            <span className={`text-sm font-medium transition-colors duration-300 ${
   338→              idx === currentStep ? 'text-brand' : idx < currentStep ? 'text-emerald-600' : 'text-muted-foreground'
   339→            }`}>
   340→              {step}
   341→            </span>
   342→            {idx === currentStep && (
   343→              <motion.div
   344→                className="ml-auto size-4 rounded-full bg-brand"
   345→                animate={{ opacity: [1, 0.3, 1] }}
   346→                transition={{ duration: 1.2, repeat: Infinity }}
   347→                aria-hidden="true"
   348→              />
   349→            )}
   350→          </motion.div>
   351→        ))}
   352→      </div>
   353→    </motion.div>
   354→  );
   355→}

   660→// ─── Suggestions Checklist ───────────────────────────────────────────────────
   661→
   662→function SuggestionsChecklist({ suggestions }: { suggestions: Suggestion[] }) {
   663→  const [checkedItems, setCheckedItems] = useState<Set<number>>(new Set());
   664→
   665→  const toggleItem = (idx: number) => {
   666→    setCheckedItems((prev) => {
   667→      const next = new Set(prev);
   668→      if (next.has(idx)) next.delete(idx);
   669→      else next.add(idx);
   670→      return next;
   671→    });
   672→  };
   673→
   674→  if (suggestions.length === 0) return null;
   675→
   676→  const sorted = [...suggestions].sort((a, b) => {
   677→    const order = { high: 0, medium: 1, low: 2 };
   678→    return order[a.priority] - order[b.priority];
   679→  });
   680→
   681→  return (
   682→    <motion.div
   683→      initial={{ opacity: 0, y: 16 }}
   684→      animate={{ opacity: 1, y: 0 }}
   685→      transition={{ delay: 0.55 }}
   686→    >
   687→      <Card>
   688→        <CardHeader className="pb-3">
   689→          <CardTitle className="flex items-center gap-2 text-base">
   690→            <Lightbulb className="size-5 text-brand" aria-hidden="true" />
   691→            Improvement Suggestions
   692→          </CardTitle>
   693→        </CardHeader>
   694→        <CardContent>
   695→          <ScrollArea className="max-h-96">
   696→            <div className="space-y-2 pr-3">
   697→              {sorted.map((item, idx) => {
   698→                const isChecked = checkedItems.has(idx);
   699→                return (
   700→                  <motion.button
   701→                    key={`${item.message}-${idx}`}
   702→                    type="button"
   703→                    onClick={() => toggleItem(idx)}
   704→                    initial={{ opacity: 0, x: -8 }}
   705→                    animate={{ opacity: 1, x: 0 }}
   706→                    transition={{ delay: 0.04 * idx, duration: 0.25 }}
   707→                    className={`flex w-full items-start gap-3 rounded-lg border p-3 text-left text-sm transition-colors ${
   708→                      isChecked
   709→                        ? 'border-emerald-200 bg-emerald-50/50 dark:border-emerald-800 dark:bg-emerald-950/40'
   710→                        : 'border-transparent bg-muted/40 hover:bg-muted/60 dark:bg-muted/30 dark:hover:bg-muted/50'
   711→                    }`}
   712→                    aria-pressed={isChecked}
   713→                  >
   714→                    <motion.div
   715→                      animate={isChecked ? { scale: [1, 1.3, 1] } : {}}
   716→                      transition={{ duration: 0.3 }}
   717→                    >
   718→                      <CircleCheckBig
   719→                        className={`size-5 shrink-0 mt-0.5 transition-colors ${
   720→                          isChecked ? 'text-emerald-500' : 'text-muted-foreground/40'
   721→                        }`}
   722→                        aria-hidden="true"
   723→                      />
   724→                    </motion.div>
   725→                    <div className="flex-1 min-w-0">
   726→                      <div className="flex flex-wrap items-center gap-1.5">
   727→                        <Badge
   728→                          variant="outline"
   729→                          className={`text-[10px] px-1.5 py-0 ${getPriorityColor(item.priority)}`}
   730→                        >
   731→                          {item.priority}
   732→                        </Badge>
   733→                        <Badge variant="secondary" className="text-[10px] px-1.5 py-0 capitalize">
   734→                          {item.section}
   735→                        </Badge>
   736→                      </div>
   737→                      <p className={`mt-1.5 leading-relaxed ${isChecked ? 'text-muted-foreground line-through' : 'text-foreground'}`}>
   738→                        {item.message}
   739→                      </p>
   740→                    </div>
   741→                  </motion.button>
   742→                );
   743→              })}
   744→            </div>
   745→          </ScrollArea>
   746→        </CardContent>
   747→      </Card>
   748→    </motion.div>
