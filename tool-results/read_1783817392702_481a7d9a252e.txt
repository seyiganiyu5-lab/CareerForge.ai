   482→function estimateYearsOfExperience(workExperience?: ResumeContext['workExperience']): string {
   483→  if (!workExperience || workExperience.length === 0) return 'several';
   484→  let totalYears = 0;
   485→  for (const exp of workExperience) {
   486→    if (exp.startDate) {
   487→      const start = new Date(exp.startDate);
   488→      const end = exp.current ? new Date() : (exp.endDate ? new Date(exp.endDate) : new Date());
   489→      const diffMs = end.getTime() - start.getTime();
   490→      totalYears += Math.max(0, Math.round(diffMs / (365.25 * 24 * 60 * 60 * 1000)));
   491→    }
   492→  }
   493→  return totalYears > 0 ? String(totalYears) : 'several';
   494→}

   520→// ---------------------------------------------------------------------------
   521→// POST Handler
   522→// ---------------------------------------------------------------------------
   523→
   524→export async function POST(
   525→  request: NextRequest,
   526→  { params }: { params: Promise<{ id: string }> }
   527→) {
   528→  try {
   529→    // 1. Authentication
   530→    const auth = await getAuthUser();
   531→    if (!auth) {
   532→      return NextResponse.json(
   533→        { success: false, error: 'Unauthorized' },
   534→        { status: 401 }
   535→      );
   536→    }
   537→
   538→    // 2. Parse & validate request body
   539→    const body: AiAssistBody = await request.json();
   540→    const { section, action, content, customPrompt, tone, resumeContext, itemId, field } = body;
   541→
   542→    if (!section || !action) {
   543→      return NextResponse.json(
   544→        { success: false, error: 'Missing required fields: section and action' },
   545→        { status: 400 }
   546→      );
   547→    }
   548→
   549→    if (!VALID_SECTIONS.includes(section as Section)) {
   550→      return NextResponse.json(
   551→        { success: false, error: `Invalid section. Must be one of: ${VALID_SECTIONS.join(', ')}` },
   552→        { status: 400 }
   553→      );
   554→    }
   555→
   556→    if (!VALID_ACTIONS.includes(action as Action)) {
   557→      return NextResponse.json(
   558→        { success: false, error: `Invalid action. Must be one of: ${VALID_ACTIONS.join(', ')}` },
   559→        { status: 400 }
   560→      );
   561→    }
   562→
   563→    if (action === 'custom' && !customPrompt) {
   564→      return NextResponse.json(
   565→        { success: false, error: 'customPrompt is required when action is "custom"' },
   566→        { status: 400 }
   567→      );
   568→    }
   569→
   570→    // 3. Resolve resume ID from route params
   571→    const { id } = await params;
   572→
   573→    // 4. Ownership check
   574→    const resume = await db.resume.findUnique({
   575→      where: { id },
   576→    });
   577→
   578→    if (!resume || resume.user_id !== auth.userId) {
   579→      return NextResponse.json(
   580→        { success: false, error: 'Resume not found' },
   581→        { status: 404 }
   582→      );
   583→    }
   584→
   585→    // 5. Feature gating
   586→    const usage = await checkAndIncrementUsage(auth.userId, FEATURE_KEYS.AI_RESUME_BUILDER);
   587→    if (!usage.allowed) {
   588→      return NextResponse.json(
   589→        {
   590→          success: false,
   591→          error: usage.reason || 'limit_reached',
   592→          limit: usage.limit,
   593→          used: usage.used,
   594→          planName: usage.planName,
   595→        },
   596→        { status: 403 }
   597→      );
   598→    }
   599→
   600→    // 6. Build the AI prompt
   601→    const { systemPrompt, userMessage } = buildPrompt(section, action, body);
   602→
   603→    // 7. Call the AI SDK
   604→    const zai = await ZAI.create();
   605→    const completion = await zai.chat.completions.create({
   606→      messages: [
   607→        { role: 'assistant', content: systemPrompt },
   608→        { role: 'user', content: userMessage },
   609→      ],
   610→      thinking: { type: 'disabled' },
   611→    });
   612→
   613→    const aiResponse = completion.choices[0]?.message?.content?.trim();
   614→
   615→    if (!aiResponse) {
   616→      return NextResponse.json(
   617→        { success: false, error: 'Failed to generate suggestion' },
   618→        { status: 500 }
   619→      );
   620→    }
   621→
   622→    // 8. Save to AiHistory
   623→    await db.aiHistory.create({
   624→      data: {
   625→        resume_id: id,
   626→        user_id: auth.userId,
   627→        section,
   628→        action,
   629→        prompt: systemPrompt + '\n\n' + userMessage,
   630→        result: aiResponse,
   631→        original_text: content || '',
   632→      },
   633→    });
   634→
   635→    // 9. Return success
   636→    return NextResponse.json({
   637→      success: true,
   638→      result: aiResponse,
   639→      action,
   640→      section,
   641→      usage: {
   642→        limit: usage.limit,
   643→        used: usage.used + 1, // reflect the just-incremented count
   644→      },
   645→    });
   646→  } catch (error) {
   647→    console.error('[AI Assist] Error:', error);
   648→
   649→    // Handle ZAI-specific errors gracefully
   650→    if (error instanceof Error) {
   651→      const message = error.message.toLowerCase();
   652→      if (
   653→        message.includes('rate limit') ||
   654→        message.includes('quota') ||
   655→        message.includes('too many requests')
   656→      ) {
   657→        return NextResponse.json(
   658→          { success: false, error: 'AI service is temporarily unavailable. Please try again in a moment.' },
   659→          { status: 503 }
   660→        );
   661→      }
   662→
   663→      if (
   664→        message.includes('api key') ||
   665→        message.includes('authentication') ||
   666→        message.includes('unauthorized')
   667→      ) {
   668→        return NextResponse.json(
   669→          { success: false, error: 'AI service configuration error. Please contact support.' },
   670→          { status: 500 }
   671→        );
   672→      }
   673→
   674→      if (
   675→        message.includes('timeout') ||
   676→        message.includes('network') ||
   677→        message.includes('fetch')
   678→      ) {
   679→        return NextResponse.json(
   680→          { success: false, error: 'AI service request timed out. Please try again.' },
   681→          { status: 504 }
   682→        );
   683→      }
   684→    }
   685→
   686→    return NextResponse.json(
   687→      { success: false, error: 'Failed to generate suggestion' },
   688→      { status: 500 }
   689→    );
   690→  }
   691→}
