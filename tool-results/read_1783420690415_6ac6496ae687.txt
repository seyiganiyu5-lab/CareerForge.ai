   381→export function searchBlogPosts(query: string, locale: Locale): BlogPost[] {
   382→  const q = query.toLowerCase().trim();
   383→  if (!q) return [];
   384→  return blogPosts.filter(p => {
   385→    const c = p.content[locale];
   386→    return (
   387→      c.title.toLowerCase().includes(q) ||
   388→      c.description.toLowerCase().includes(q) ||
   389→      c.body.some(b => {
   390→        if (b.type === 'paragraph' || b.type === 'quote') return b.text.toLowerCase().includes(q);
   391→        if (b.type === 'list') return b.items.some(i => i.toLowerCase().includes(q));
   392→        if (b.type === 'heading') return b.text.toLowerCase().includes(q);
   393→        if (b.type === 'callout') return (b.title + b.text).toLowerCase().includes(q);
   394→        return false;
   395→      })
   396→    );
   397→  });
