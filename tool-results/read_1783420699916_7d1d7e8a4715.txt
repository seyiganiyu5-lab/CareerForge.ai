   590→   590→export function searchArticles(query: string, locale: Locale): HelpArticle[] {
   591→   591→  const q = query.toLowerCase().trim();
   592→   592→  if (!q) return [];
   593→   593→  return helpArticles.filter(a => {
   594→   594→    const c = a.content[locale];
   595→   595→    return (
   596→   596→      c.title.toLowerCase().includes(q) ||
   597→   597→      c.description.toLowerCase().includes(q) ||
   598→   598→      c.body.some(b => {
   599→   599→        if (b.type === 'paragraph' || b.type === 'quote') return b.text.toLowerCase().includes(q);
   600→   600→        if (b.type === 'list') return b.items.some(i => i.toLowerCase().includes(q));
   601→   601→        if (b.type === 'heading') return b.text.toLowerCase().includes(q);
   602→   602→        if (b.type === 'callout') return (b.title + b.text).toLowerCase().includes(q);
   603→   603→        if (b.type === 'steps') return b.steps.some(s => (s.title + s.description).toLowerCase().includes(q));
   604→   604→        return false;
   605→   605→      })
   606→   606→    );
   607→   607→  });
   608→   608→}
